\begin{theorem*}[Sign of a Reciprocal on $\mathbb{R}$]
For every nonzero $\alpha\in\mathbb{R}$,
\[
0_{\mathbb{R}}<\alpha \Longrightarrow 0_{\mathbb{R}}<\alpha^{-1},
\qquad
\alpha<0_{\mathbb{R}} \Longrightarrow \alpha^{-1}<0_{\mathbb{R}}.
\]
\end{theorem*}
