\begin{definitionbox}{Definition}
\begin{definition}[Nonnegative Cut]
\label{def:nonnegative-cut}
A cut $\alpha\in\mathbb{R}$ is nonnegative if $0_{\mathbb{R}}\le\alpha$ and
positive if $0_{\mathbb{R}}<\alpha$.
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
For every $\alpha\in\mathbb{R}$,
\[
\alpha\ge 0_{\mathbb{R}}\Longleftrightarrow 0_{\mathbb{R}}\le\alpha,
\qquad
\alpha>0_{\mathbb{R}}\Longleftrightarrow 0_{\mathbb{R}}<\alpha.
\]
\end{remark*}

\begin{remark*}[Interpretation]
The nonnegative and positive real cuts are the cuts lying weakly or strictly
above the zero cut.
\end{remark*}

\begin{definitionbox}{Definition}
\begin{definition}[Product of Nonnegative Cuts]
\label{def:product-nonnegative-on-r}
For nonnegative cuts $\alpha,\beta\in\mathbb{R}$, define
\[
\alpha\cdot\beta
:=
\{r\in\mathbb{Q}:r<0_{\mathbb{Q}}\}
\cup
\{p\cdot q:p\in\alpha,\ q\in\beta,\ p\ge 0_{\mathbb{Q}},\ q\ge 0_{\mathbb{Q}}\}.
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
For all nonnegative cuts $\alpha,\beta\in\mathbb{R}$,
\[
\alpha\cdot\beta
=
\{r\in\mathbb{Q}:r<0_{\mathbb{Q}}\}
\cup
\{p\cdot q:p\in\alpha,\ q\in\beta,\ p\ge 0_{\mathbb{Q}},\ q\ge 0_{\mathbb{Q}}\}.
\]
\end{remark*}

\begin{remark*}[Interpretation]
The product of nonnegative cuts contains all negative rationals and all
products of nonnegative rational elements drawn from the two cuts.
\end{remark*}

\begin{definitionbox}{Definition}
\begin{definition}[Absolute Value on $\mathbb{R}$]
\label{def:real-chapter-absolute-value-on-r}
For $\alpha\in\mathbb{R}$, define
\[
|\alpha|:=
\begin{cases}
\alpha, & 0_{\mathbb{R}}\le\alpha,\\
-\alpha, & \alpha<0_{\mathbb{R}}.
\end{cases}
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
For every $\alpha\in\mathbb{R}$,
\[
0_{\mathbb{R}}\le\alpha\Longrightarrow |\alpha|=\alpha,
\qquad
\alpha<0_{\mathbb{R}}\Longrightarrow |\alpha|=-\alpha.
\]
\end{remark*}

\begin{remark*}[Interpretation]
The absolute value of a real cut is the cut itself when nonnegative and its
additive inverse when negative.
\end{remark*}

\begin{definitionbox}{Definition}
\begin{definition}[Multiplication of Cuts]
\label{def:multiplication-on-r}
For cuts $\alpha,\beta\in\mathbb{R}$, define $\alpha\cdot\beta$ by sign cases:
use $|\alpha|\cdot|\beta|$ when $\alpha$ and $\beta$ have the same sign, use
$-(|\alpha|\cdot|\beta|)$ when they have opposite signs, and use
$0_{\mathbb{R}}$ when either factor is $0_{\mathbb{R}}$.
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
For all $\alpha,\beta\in\mathbb{R}$, the product $\alpha\cdot\beta$ is defined
by the sign cases determined by $|\alpha|$, $|\beta|$, and $0_{\mathbb{R}}$.
\end{remark*}

\begin{remark*}[Interpretation]
Real multiplication extends the nonnegative cut product to all signs by using
absolute values, negation, and the zero case.
\end{remark*}

\Longrightarrow
\beta^{-1}\le\alpha^{-1}<0_{\mathbb{R}}.
\]
\hyperref[prf:reciprocal-reverses-nonstrict-order-on-same-sign-r]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
For all nonzero $\alpha,\beta\in\mathbb{R}$,
\[
0_{\mathbb{R}}<\alpha\le\beta
\Longrightarrow
0_{\mathbb{R}}<\beta^{-1}\le\alpha^{-1},
\qquad
\alpha\le\beta<0_{\mathbb{R}}
\Longrightarrow
\beta^{-1}\le\alpha^{-1}<0_{\mathbb{R}}.
\]
\end{remark*}

\begin{remark*}[Interpretation]
For same-sign nonzero real numbers, reciprocal reverses non-strict order and
preserves the common sign side.
\end{remark*}

|\alpha^{-1}|=|\alpha|^{-1}.

|\alpha^{-1}|=|\alpha|^{-1}.

\begin{remark*}[Interpretation]
Absolute value commutes with reciprocals of nonzero real numbers.

\begin{definitionbox}{Definition}
\begin{definition}[Division on $\mathbb{R}$]
\label{def:division-on-r}
For $\beta\ne0_{\mathbb{R}}$, define
\[
\alpha/\beta:=\alpha\cdot\beta^{-1}.
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
For all $\alpha\in\mathbb{R}$ and all
$\beta\in\mathbb{R}\setminus\{0_{\mathbb{R}}\}$,
\[
\alpha/\beta=\alpha\cdot\beta^{-1}.
\]
\end{remark*}

\begin{remark*}[Interpretation]
Real division is multiplication by the reciprocal of the nonzero divisor.

\begin{theorembox}{Theorem}
\begin{theorem}[Distributivity on $\mathbb{R}$]
\label{thm:distributivity-on-r}
For all $\alpha,\beta,\gamma\in\mathbb{R}$,
\[
\alpha\cdot(\beta+\gamma)=\alpha\cdot\beta+\alpha\cdot\gamma.
\]
\hyperref[prf:distributivity-on-r]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\operatorname{LeftDistributive}(\mathbb{R},\cdot,+).
\]
\end{remark*}

\begin{remark*}[Interpretation]
Real multiplication distributes over real addition on the left.
\end{remark*}
