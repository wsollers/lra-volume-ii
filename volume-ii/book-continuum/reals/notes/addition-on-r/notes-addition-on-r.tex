\begin{definitionbox}{Definition}
\begin{definition}[Addition of Cuts]
\label{def:addition-on-r}
For cuts $\alpha,\beta\in\mathbb{R}$, define
\[
\alpha+\beta:=\{p+q:p\in\alpha,\ q\in\beta\}.
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
For all cuts $\alpha,\beta\in\mathbb{R}$,
\[
\alpha+\beta=\{p+q:p\in\alpha,\ q\in\beta\}.
\]
\end{remark*}

\begin{remark*}[Interpretation]
Addition of real cuts is defined pointwise from rational sums of elements in
the two cuts.
\end{remark*}

\begin{lemmabox}{Lemma}
\begin{lemma}[Sum of Cuts Is a Cut]
\label{lem:sum-of-cuts-is-a-cut}
If $\alpha,\beta\in\mathbb{R}$, then $\alpha+\beta$ is a Dedekind cut.
\hyperref[prf:sum-of-cuts-is-a-cut]{\textit{Go to proof.}}
\end{lemma}
\end{lemmabox}

\begin{remark*}[Standard quantified statement]
For all $\alpha,\beta\in\mathbb{R}$,
\[
\alpha+\beta\in\mathbb{R}.
\]
\end{remark*}

\begin{remark*}[Interpretation]
The pointwise sum of two Dedekind cuts is again a Dedekind cut.
\end{remark*}

\label{thm:addition-on-r-associative}

(\alpha+\beta)+\gamma=\alpha+(\beta+\gamma).

\begin{definitionbox}{Definition}
\begin{definition}[Negation of a Cut]
\label{def:negation-on-r}
For $\alpha\in\mathbb{R}$, define
\[
-\alpha:=\{p\in\mathbb{Q}:\exists r\in\mathbb{Q}\,(r>0\land -p-r\notin\alpha)\}.
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
For every $\alpha\in\mathbb{R}$,
\[
-\alpha=\{p\in\mathbb{Q}:\exists r\in\mathbb{Q}\,(r>0\land -p-r\notin\alpha)\}.
\]
\end{remark*}

\begin{remark*}[Interpretation]
The negative cut consists of rationals lying safely below the reflected
complement of $\alpha$.
\end{remark*}

\begin{theorembox}{Theorem}
\begin{theorem}[$\mathbb{R}$ Is an Additive Abelian Group]
\label{thm:r-additive-abelian-group}
The structure
 $(\mathbb{R},+,0_{\mathbb{R}})$ is an abelian group.
\hyperref[prf:r-additive-abelian-group]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\operatorname{IsAbelianGroup}
\bigl(\operatorname{AbelianGroup}(
\operatorname{Group}(\operatorname{Monoid}(
\operatorname{Semigroup}(\operatorname{Magma}(\mathbb{R},+)),
0_{\mathbb{R}})))\bigr).
\]
\end{remark*}

\begin{remark*}[Interpretation]
The real numbers under addition, with zero as identity, form an abelian group.
\end{remark*}
