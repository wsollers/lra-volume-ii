\begin{definitionbox}{Definition}
\begin{definition}[Upper Bound in $\mathbb{R}$]
\label{def:upper-bound-in-r}
Let $S\subseteq\mathbb{R}$. A cut $\gamma\in\mathbb{R}$ is an upper bound of $S$
if
\[
\forall\alpha\in S\,(\alpha\le\gamma).
\]
The set $S$ is bounded above if it has an upper bound in $\mathbb{R}$.
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\operatorname{UpperBound}(\gamma,S,\mathbb{R},\le)
\Longleftrightarrow
\gamma\in\mathbb{R}\land S\subseteq\mathbb{R}\land
\forall\alpha\in S\,(\alpha\le\gamma),
\]
\[
\operatorname{BoundedAbove}(S,\mathbb{R},\le)
\Longleftrightarrow
\exists\gamma\in\mathbb{R}\,\operatorname{UpperBound}(\gamma,S,\mathbb{R},\le).
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{UpperBound}(\gamma,S,\mathbb{R},\le)
\]
means that $\gamma$ is at least every element of $S$ in the real order.
\end{remark*}

\begin{remark*}[Interpretation]
An upper bound is a real number that lies above all elements of the given real subset.
\end{remark*}

\begin{definitionbox}{Definition}
\begin{definition}[Supremum in $\mathbb{R}$]
\label{def:supremum-in-r}
For $S\subseteq\mathbb{R}$, a cut $\sigma$ is the supremum of $S$ if $\sigma$
is an upper bound of $S$ and every upper bound $\gamma$ of $S$ satisfies
$\sigma\le\gamma$.
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\operatorname{LeastUpperBound}(\sigma,S,\mathbb{R},\le)
\Longleftrightarrow
\operatorname{UpperBound}(\sigma,S,\mathbb{R},\le)
\land
\forall\gamma\in\mathbb{R}\,
\left(\operatorname{UpperBound}(\gamma,S,\mathbb{R},\le)\Longrightarrow \sigma\le\gamma\right).
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{LeastUpperBound}(\sigma,S,\mathbb{R},\le)
\]
means that $\sigma$ is the least real upper bound of $S$.
\end{remark*}

\begin{remark*}[Interpretation]
The supremum of a set is its smallest real upper bound.
\end{remark*}

\begin{definitionbox}{Definition}
\begin{definition}[Least-Upper-Bound Property on $\mathbb{R}$]
\label{def:least-upper-bound-property-on-r}
The ordered set $\mathbb{R}$ has the \emph{least-upper-bound property} if every
nonempty subset of $\mathbb{R}$ that is bounded above has a supremum in
$\mathbb{R}$.
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\operatorname{OrderComplete}(\mathbb{R},\le)
\Longleftrightarrow
\forall S\subseteq\mathbb{R},\,
\left(S\ne\emptyset\land \operatorname{BoundedAbove}(S,\mathbb{R},\le)\right)
\Longrightarrow
\exists\sigma\in\mathbb{R}\,\operatorname{LeastUpperBound}(\sigma,S,\mathbb{R},\le).
\]
\end{remark*}

\begin{remark*}[Interpretation]
Order completeness asserts that every nonempty bounded-above real set has a real supremum.
\end{remark*}

\begin{lemmabox}{Lemma}
\begin{lemma}[Union of a Bounded Nonempty Family of Cuts Is a Cut]
\label{lem:union-of-cuts-is-a-cut}
If $S\subseteq\mathbb{R}$ is nonempty and bounded above, then
\[
\bigcup_{\alpha\in S}\alpha
\]
is a Dedekind cut.
\hyperref[prf:union-of-cuts-is-a-cut]{\textit{Go to proof.}}
\end{lemma}
\end{lemmabox}

\begin{remark*}[Standard quantified statement]
\[
\forall S\subseteq\mathbb{R},\qquad
\left(S\ne\emptyset\land \operatorname{BoundedAbove}(S,\mathbb{R},\le)\right)
\Longrightarrow
\operatorname{DedekindCut}\!\left(\bigcup_{\alpha\in S}\alpha\right).
\]
\end{remark*}

\begin{remark*}[Interpretation]
The union of a nonempty bounded-above family of real cuts is itself a real cut.
\end{remark*}

\begin{theorembox}{Theorem}
\begin{theorem}[Least-Upper-Bound Property]
\label{thm:lub-property-r}
Every nonempty subset $S\subseteq\mathbb{R}$ that is bounded above has a least
upper bound, namely
\[
\sup S=\bigcup_{\alpha\in S}\alpha.
\]
\hyperref[prf:lub-property-r]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\forall S\subseteq\mathbb{R},\qquad
\left(S\ne\emptyset\land \operatorname{BoundedAbove}(S,\mathbb{R},\le)\right)
\Longrightarrow
\operatorname{LeastUpperBound}\!\left(\bigcup_{\alpha\in S}\alpha,S,\mathbb{R},\le\right).
\]
\end{remark*}

\begin{remark*}[Interpretation]
For Dedekind cuts, the supremum of a nonempty bounded-above family is the union of its member cuts.
\end{remark*}

\begin{corollarybox}{Corollary}
\begin{corollary}[Greatest-Lower-Bound Property]
\label{cor:glb-property-r}
Every nonempty subset $S\subseteq\mathbb{R}$ that is bounded below has a
greatest lower bound.
\hyperref[prf:glb-property-r]{\textit{Go to proof.}}
\end{corollary}
\end{corollarybox}

\begin{remark*}[Standard quantified statement]
\[
\forall S\subseteq\mathbb{R},\qquad
\left(S\ne\emptyset\land \operatorname{BoundedBelow}(S,\mathbb{R},\le)\right)
\Longrightarrow
\exists\lambda\in\mathbb{R}\,\operatorname{GreatestLowerBound}(\lambda,S,\mathbb{R},\le).
\]
\end{remark*}

\begin{remark*}[Interpretation]
The least-upper-bound property also yields greatest lower bounds for nonempty bounded-below real sets.
\end{remark*}
