\subsection*{Order by Inclusion}

\begin{definitionbox}{Definition}
\begin{definition}[Order on $\mathbb{R}$]
\label{def:order-on-r}
For cuts $\alpha,\beta\in\mathbb{R}$, define
\[
\alpha\le\beta \quad:\Longleftrightarrow\quad \alpha\subseteq\beta.
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\forall \alpha,\beta\in\mathbb{R},\qquad
\alpha\le\beta \Longleftrightarrow \alpha\subseteq\beta.
\]
\end{remark*}

\begin{remark*}[Interpretation]
The nonstrict order on real numbers is the inclusion order on the Dedekind cuts representing those real numbers.
\end{remark*}

\begin{definitionbox}{Definition}
\begin{definition}[Strict Order on $\mathbb{R}$]
\label{def:strict-order-on-r}
For cuts $\alpha,\beta\in\mathbb{R}$, define
\[
\alpha<\beta \quad:\Longleftrightarrow\quad \alpha\subsetneq\beta.
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\forall \alpha,\beta\in\mathbb{R},\qquad
\alpha<\beta \Longleftrightarrow \alpha\subsetneq\beta.
\]
\end{remark*}

\begin{remark*}[Interpretation]
Strict order on real numbers records proper inclusion of cuts.
\end{remark*}

\begin{theorembox}{Theorem}
\begin{theorem}[Order on $\mathbb{R}$ Is Reflexive]
\label{thm:order-on-r-reflexive}
For every $\alpha\in\mathbb{R}$, $\alpha\le\alpha$.
\hyperref[prf:order-on-r-reflexive]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\forall \alpha\in\mathbb{R},\qquad \alpha\le\alpha.
\]
\end{remark*}

\begin{remark*}[Interpretation]
Every real cut is included in itself, so the induced order is reflexive.
\end{remark*}

\begin{theorembox}{Theorem}
\begin{theorem}[Order on $\mathbb{R}$ Is Antisymmetric]
\label{thm:order-on-r-antisymmetric}
For all $\alpha,\beta\in\mathbb{R}$,
\[
\alpha\le\beta\land\beta\le\alpha \Longrightarrow \alpha=\beta.
\]
\hyperref[prf:order-on-r-antisymmetric]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\forall \alpha,\beta\in\mathbb{R},\qquad
(\alpha\le\beta\land\beta\le\alpha)\Longrightarrow \alpha=\beta.
\]
\end{remark*}

\begin{remark*}[Interpretation]
Two real cuts that include each other have exactly the same rational elements, hence are the same real number.
\end{remark*}

\begin{theorembox}{Theorem}
\begin{theorem}[Order on $\mathbb{R}$ Is Transitive]
\label{thm:order-on-r-transitive}
For all $\alpha,\beta,\gamma\in\mathbb{R}$,
\[
\alpha\le\beta\land\beta\le\gamma \Longrightarrow \alpha\le\gamma.
\]
\hyperref[prf:order-on-r-transitive]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\forall \alpha,\beta,\gamma\in\mathbb{R},\qquad
(\alpha\le\beta\land\beta\le\gamma)\Longrightarrow \alpha\le\gamma.
\]
\end{remark*}

\begin{remark*}[Interpretation]
Inclusion of cuts composes, so the real order is transitive.
\end{remark*}

\begin{theorembox}{Theorem}
\begin{theorem}[Comparability of Cuts]
\label{thm:comparability-of-cuts}
For all cuts $\alpha,\beta\in\mathbb{R}$,
\[
\alpha\subseteq\beta \lor \beta\subseteq\alpha.
\]
\hyperref[prf:comparability-of-cuts]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\forall \alpha,\beta\in\mathbb{R},\qquad
\alpha\subseteq\beta \lor \beta\subseteq\alpha.
\]
\end{remark*}

\begin{remark*}[Interpretation]
Any two Dedekind cuts are comparable by inclusion.
\end{remark*}

\begin{theorembox}{Theorem}
\begin{theorem}[Order on $\mathbb{R}$ Is Total]
\label{thm:order-on-r-total}
For all $\alpha,\beta\in\mathbb{R}$,
\[
\alpha\le\beta\lor\beta\le\alpha.
\]
\hyperref[prf:order-on-r-total]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\forall \alpha,\beta\in\mathbb{R},\qquad
\alpha\le\beta \lor \beta\le\alpha.
\]
\end{remark*}

\begin{remark*}[Interpretation]
Because the real order is inclusion of cuts, comparability of cuts gives totality of the real order.
\end{remark*}

\begin{theorembox}{Theorem}
\begin{theorem}[Trichotomy on $\mathbb{R}$]
\label{thm:trichotomy-on-r}
For all $\alpha,\beta\in\mathbb{R}$, exactly one of
\[
\alpha<\beta,\qquad \alpha=\beta,\qquad \beta<\alpha
\]
holds.
\hyperref[prf:trichotomy-on-r]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\forall \alpha,\beta\in\mathbb{R},\qquad
\operatorname{ExactlyOne}(\alpha<\beta,\alpha=\beta,\beta<\alpha).
\]
\end{remark*}

\begin{remark*}[Interpretation]
Two real numbers have exactly one strict order relation between them, or else they are equal.
\end{remark*}

\begin{theorembox}{Theorem}
\begin{theorem}[$\mathbb{R}$ Is a Totally Ordered Set]
\label{thm:r-totally-ordered-set}
The ordered pair $(\mathbb{R},\le)$ is a total order.
\hyperref[prf:r-totally-ordered-set]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\operatorname{TotalOrder}(\mathbb{R},\le).
\]
\end{remark*}

\begin{remark*}[Interpretation]
The real numbers, ordered by inclusion of cuts, form a totally ordered set.
\end{remark*}
