\begin{theorembox}{Theorem (Existence of Square Roots)}
\begin{theorem}[Existence of Square Roots]
\label{thm:existence-of-square-roots}
For every real number $a\ge 0$, there exists $x\in\mathbb{R}$ such that
\[
x\ge 0
\quad\text{and}\quad
x^2=a.
\]
\hyperref[prf:existence-of-square-roots]{Go to proof.}
\end{theorem}
\end{theorembox}
\begin{remark*}[Standard quantified statement]
\[
\forall a\in\mathbb{R}\;
\bigl(a\ge 0\Rightarrow
\exists x\in\mathbb{R}\;(x\ge 0\land x^2=a)\bigr).
\]
\end{remark*}
\begin{remark*}[Predicate reading]
\[
\operatorname{Nonnegative}(a)\Rightarrow
\operatorname{HasNonnegativeSquareRoot}(a).
\]
\end{remark*}
\begin{remark*}[Negated quantified statement]
\[
\exists a\in\mathbb{R}\;
\bigl(a\ge 0\land
\forall x\in\mathbb{R}\;(x<0\lor x^2\ne a)\bigr).
\]
\end{remark*}
\begin{remark*}[Negation predicate reading]
\[
\operatorname{Nonnegative}(a)\land
\neg\operatorname{HasNonnegativeSquareRoot}(a).
\]
\end{remark*}

\begin{remark*}[Failure modes]
\begin{description}
  \item[Exposition.] The named branch below records the displayed negation of the predicate.
  \item[\(\operatorname{Nonnegative}\) fails.] The predicate fails exactly in the displayed negation case:
  \[
  \exists a\in\mathbb{R}\;
  \bigl(a\ge 0\land
  \forall x\in\mathbb{R}\;(x<0\lor x^2\ne a)\bigr).
  \]
  In predicate-reading form this is recorded as
  \[
  \operatorname{Nonnegative}(a)\land
  \neg\operatorname{HasNonnegativeSquareRoot}(a).
  \]
\end{description}
\end{remark*}
\begin{remark*}[Interpretation]
Nonnegative real numbers have nonnegative real square roots. This is a
completeness consequence, not an algebraic field axiom.
\end{remark*}

\hyperref[cor:ordered-field-laws-on-r]{Ordered-Field Laws Hold on $\mathbb{R}$}.
\end{dependencies}

\begin{theorembox}{Theorem (Uniqueness of Nonnegative Square Roots)}
\begin{theorem}[Uniqueness of Nonnegative Square Roots]
\label{thm:uniqueness-of-nonnegative-square-roots}
If $x,y\ge 0$ and
\[
x^2=y^2,
\]
then
\[
x=y.
\]
\hyperref[prf:uniqueness-of-nonnegative-square-roots]{Go to proof.}
\end{theorem}
\end{theorembox}
\begin{remark*}[Standard quantified statement]
\[
\forall x,y\in\mathbb{R}\;
\bigl((x\ge 0\land y\ge 0\land x^2=y^2)\Rightarrow x=y\bigr).
\]
\end{remark*}
\begin{remark*}[Predicate reading]
\[
\operatorname{Nonnegative}(x)\land\operatorname{Nonnegative}(y)\land x^2=y^2
\Rightarrow x=y.
\]
\end{remark*}
\begin{remark*}[Negated quantified statement]
\[
\exists x,y\in\mathbb{R}\;
\bigl(x\ge 0\land y\ge 0\land x^2=y^2\land x\ne y\bigr).
\]
\end{remark*}
\begin{remark*}[Negation predicate reading]
\[
\operatorname{SameSquare}(x,y)\land\operatorname{Nonnegative}(x,y)\land x\ne y.
\]
\end{remark*}

\begin{remark*}[Failure modes]
\begin{description}
  \item[Exposition.] The named branch below records the displayed negation of the predicate.
  \item[\(\operatorname{Nonnegative}\) fails.] The predicate fails exactly in the displayed negation case:
  \[
  \exists x,y\in\mathbb{R}\;
  \bigl(x\ge 0\land y\ge 0\land x^2=y^2\land x\ne y\bigr).
  \]
  In predicate-reading form this is recorded as
  \[
  \operatorname{SameSquare}(x,y)\land\operatorname{Nonnegative}(x,y)\land x\ne y.
  \]
\end{description}
\end{remark*}
\begin{remark*}[Interpretation]
The sign restriction removes the ambiguity between opposite square roots.
\end{remark*}

\begin{definitionbox}{Definition (Square Root)}
\begin{definition}[Square Root]
\label{def:square-root}
For $a\ge 0$, define $\sqrt a$ to be the unique nonnegative real number $x$
such that
\[
x^2=a.
\]
\end{definition}
\end{definitionbox}
\begin{remark*}[Standard quantified statement]
\[
a\ge 0
\Rightarrow
\sqrt a=\text{the unique }x\in\mathbb{R}\text{ with }x\ge 0\text{ and }x^2=a.
\]
\end{remark*}
\begin{remark*}[Predicate reading]
\[
\sqrt a=\iota x\;\operatorname{NonnegativeSquareRoot}(x,a).
\]
\end{remark*}
\begin{remark*}[Negated quantified statement]
\[
a\ge 0\land
\neg\exists!x\in\mathbb{R}\;(x\ge 0\land x^2=a).
\]
\end{remark*}
\begin{remark*}[Negation predicate reading]
\[
\neg\operatorname{UniqueNonnegativeSquareRoot}(a).
\]
\end{remark*}

\begin{remark*}[Failure modes]
\begin{description}
  \item[Exposition.] The named branch below records the displayed negation of the predicate.
  \item[\(\operatorname{NonnegativeSquareRoot}\) fails.] The predicate fails exactly in the displayed negation case:
  \[
  a\ge 0\land
  \neg\exists!x\in\mathbb{R}\;(x\ge 0\land x^2=a).
  \]
  In predicate-reading form this is recorded as
  \[
  \neg\operatorname{UniqueNonnegativeSquareRoot}(a).
  \]
\end{description}
\end{remark*}
\begin{remark*}[Interpretation]
The radical symbol names the unique nonnegative solution of the square
equation.
\end{remark*}

\begin{theorembox}{Theorem (Square Root of Two Is Irrational)}
\begin{theorem}[Square Root of Two Is Irrational]
\label{thm:square-root-of-two-is-irrational}
\[
\sqrt2\in\mathbb{R}\setminus\mathbb{Q}.
\]
\hyperref[prf:square-root-of-two-is-irrational]{Go to proof.}
\end{theorem}
\end{theorembox}
\begin{remark*}[Standard quantified statement]
\[
\sqrt2\in\mathbb{R}\land \sqrt2\notin\mathbb{Q}.
\]
\end{remark*}
\begin{remark*}[Predicate reading]
\[
\operatorname{Irrational}(\sqrt2).
\]
\end{remark*}
\begin{remark*}[Negated quantified statement]
\[
\sqrt2\in\mathbb{Q}.
\]
\end{remark*}
\begin{remark*}[Negation predicate reading]
\[
\neg\operatorname{Irrational}(\sqrt2).
\]
\end{remark*}

\begin{remark*}[Failure modes]
\begin{description}
  \item[Exposition.] The named branch below records the displayed negation of the predicate.
  \item[\(\operatorname{Irrational}\) fails.] The predicate fails exactly in the displayed negation case:
  \[
  \sqrt2\in\mathbb{Q}.
  \]
  In predicate-reading form this is recorded as
  \[
  \neg\operatorname{Irrational}(\sqrt2).
  \]
\end{description}
\end{remark*}
\begin{remark*}[Interpretation]
The square root supplied by completeness need not be rational; the classical
gap at $2$ is filled only after passing from $\mathbb{Q}$ to $\mathbb{R}$.
\end{remark*}

\begin{definitionbox}{Definition}
\begin{definition}[The Square-Root-of-Two Cut]
\label{def:sqrt-two-cut}
Define
\[
\sqrt{2}:=\{r\in\mathbb{Q}:r<0_{\mathbb{Q}}\lor r\cdot r<2_{\mathbb{Q}}\}.
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\sqrt{2}=\{r\in\mathbb{Q}:r<0_{\mathbb{Q}}\lor r\cdot r<2_{\mathbb{Q}}\}.
\]
\end{remark*}

\begin{remark*}[Interpretation]
The square-root-of-two cut collects negative rationals together with positive rationals whose square is below two.
\end{remark*}

\begin{theorembox}{Theorem}
\begin{theorem}[Its Square Is Two]
\label{thm:sqrt-two-squared-is-two}
In $\mathbb{R}$,
\[
\sqrt{2}\cdot\sqrt{2}=2_{\mathbb{R}},
\]
where $2_{\mathbb{R}}:=\iota(2_{\mathbb{Q}})$.
\hyperref[prf:sqrt-two-squared-is-two]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\sqrt{2}\cdot\sqrt{2}=2_{\mathbb{R}}
\quad\text{with}\quad
2_{\mathbb{R}}=\iota(2_{\mathbb{Q}}).
\]
\end{remark*}

\begin{remark*}[Interpretation]
The cut constructed for $\sqrt{2}$ has square equal to the real number two.
\end{remark*}

\begin{corollarybox}{Corollary}
\begin{corollary}[$\mathbb{R}$ Contains the Supremum $\mathbb{Q}$ Was Missing]
\label{cor:r-fills-the-gap}
Let
\[
S=\{q\in\mathbb{Q}:0_{\mathbb{Q}}<q\land q\cdot q<2_{\mathbb{Q}}\}.
\]
Then
\[
\sqrt{2}=\sup \iota(S).
\]
Thus the bounded-above rational set with no rational supremum has a supremum
in $\mathbb{R}$.
\hyperref[prf:r-fills-the-gap]{\textit{Go to proof.}}
\end{corollary}
\end{corollarybox}

\begin{remark*}[Standard quantified statement]
\[
S=\{q\in\mathbb{Q}:0_{\mathbb{Q}}<q\land q\cdot q<2_{\mathbb{Q}}\}
\Longrightarrow
\operatorname{LeastUpperBound}(\sqrt{2},\iota(S),\mathbb{R},\le).
\]
\end{remark*}

\begin{remark*}[Interpretation]
The real number $\sqrt{2}$ is exactly the supremum in $\mathbb{R}$ of the rational gap set embedded into the reals.
\end{remark*}
