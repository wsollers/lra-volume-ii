\subsection*{Order Compatibility}

\begin{theorembox}{Theorem}
\begin{theorem}[Addition Preserves Order on $\mathbb{R}$]
\label{thm:addition-preserves-order-on-r}
For all $\alpha,\beta,\gamma\in\mathbb{R}$,
\[
\alpha<\beta \Longrightarrow \alpha+\gamma<\beta+\gamma,
\]
and the corresponding non-strict statement also holds.
\hyperref[prf:addition-preserves-order-on-r]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\forall \alpha,\beta,\gamma\in\mathbb{R},\qquad
\alpha<\beta\Longrightarrow \alpha+\gamma<\beta+\gamma,
\]
\[
\forall \alpha,\beta,\gamma\in\mathbb{R},\qquad
\alpha\le\beta\Longrightarrow \alpha+\gamma\le\beta+\gamma.
\]
\end{remark*}

\begin{remark*}[Interpretation]
Translating both real numbers by the same real number preserves strict and nonstrict order.
\end{remark*}

\begin{theorembox}{Theorem}
\begin{theorem}[Multiplication by Positives Preserves Order on $\mathbb{R}$]
\label{thm:positive-multiplication-preserves-order-on-r}
For all $\alpha,\beta,\gamma\in\mathbb{R}$,
\[
0_{\mathbb{R}}<\gamma\land\alpha<\beta
\Longrightarrow
\gamma\cdot\alpha<\gamma\cdot\beta.
\]
\hyperref[prf:positive-multiplication-preserves-order-on-r]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\forall \alpha,\beta,\gamma\in\mathbb{R},\qquad
(0_{\mathbb{R}}<\gamma\land\alpha<\beta)
\Longrightarrow
\gamma\cdot\alpha<\gamma\cdot\beta.
\]
\end{remark*}

\begin{remark*}[Interpretation]
Multiplication by a positive real number preserves strict order.
\end{remark*}
