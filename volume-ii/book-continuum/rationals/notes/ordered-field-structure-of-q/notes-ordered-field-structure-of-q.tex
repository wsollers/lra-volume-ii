\subsection*{Ordered Field Instantiation}

\begin{theorembox}{Theorem}
\begin{theorem}[$\mathbb{Q}$ Is an Ordered Field]
\label{thm:q-is-an-ordered-field}
The structure
\[
(\mathbb{Q},+,\cdot,0_{\mathbb{Q}},1_{\mathbb{Q}},\le)
\]
is an ordered field.
\hyperref[prf:q-is-an-ordered-field]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\operatorname{IsOrderedField}(\mathbb{Q},+,\cdot,0_{\mathbb{Q}},1_{\mathbb{Q}},\le).
\]
\end{remark*}

\begin{remark*}[Interpretation]
The constructed rational operations and order make $\mathbb{Q}$ an ordered field.
\end{remark*}

\begin{corollarybox}{Corollary}
\begin{corollary}[Ordered Field Laws Hold on $\mathbb{Q}$]
\label{cor:ordered-field-laws-on-q}
Every theorem that depends only on the ordered-field axioms applies to
$\mathbb{Q}$ with its constructed operations and order.
\hyperref[prf:ordered-field-laws-on-q]{\textit{Go to proof.}}
\end{corollary}
\end{corollarybox}

\]
\end{remark*}

\begin{remark*}[Interpretation]
Every generic ordered-field consequence applies to the constructed rational ordered field.
\end{remark*}

\begin{corollary}[Squares Are Nonnegative on $\mathbb{Q}$]
\label{cor:squares-nonnegative-on-q}
For every $x\in\mathbb{Q}$,
\[
0_{\mathbb{Q}}\le x^2.
\]
\hyperref[prf:squares-nonnegative-on-q]{\textit{Go to proof.}}
\end{corollary}
\end{corollarybox}

\begin{remark*}[Standard quantified statement]
\[
\forall x\in\mathbb{Q},\qquad 0_{\mathbb{Q}}\le x^2.
\]
\end{remark*}

\begin{remark*}[Interpretation]
Every rational square is nonnegative in the rational order.
\end{remark*}

\begin{corollarybox}{Corollary}
\begin{corollary}[Fraction Comparison in $\mathbb{Q}$]
\label{cor:fraction-comparison-on-q}
For rational classes with positive denominator product,
\[
[a,b]<[c,d]
\quad\Longleftrightarrow\quad
a\cdot d<c\cdot b.
\]
\hyperref[prf:fraction-comparison-on-q]{\textit{Go to proof.}}
\end{corollary}
\end{corollarybox}

\begin{remark*}[Standard quantified statement]
\[
\forall [a,b],[c,d]\in\mathbb{Q},\qquad
0_{\mathbb{Q}}<[b,d]
\Longrightarrow
\left([a,b]<[c,d]\Longleftrightarrow a\cdot d<c\cdot b\right).
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{FractionComparison}([a,b],[c,d])
\]
means that, when the denominator product is positive, the order of the rational classes is equivalent to the corresponding cross-product inequality.
\end{remark*}

\begin{remark*}[Interpretation]
Fractions with positive denominator product can be compared by cross multiplication.
\end{remark*}
