\subsection*{No Rational Square Root of Two}

\begin{definitionbox}{Definition}
\begin{definition}[Rational Square]
\label{def:rational-square}
A rational number $a\in\mathbb{Q}$ is a \emph{rational square} if there
exists $q\in\mathbb{Q}$ such that
\[
a=q^2.
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\text{$a$ is a rational square}
\Longleftrightarrow
\exists q\in\mathbb{Q}\,(a=q^2).
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\text{$a$ is a rational square}
\]
means that $a$ lies in the image of the squaring map on $\mathbb{Q}$.
\end{remark*}

\begin{remark*}[Interpretation]
Rational squares are exactly the rational numbers that already have a rational
number whose square gives them.
\end{remark*}

\begin{theorembox}{Theorem}
\begin{theorem}[Existence of Rational Square Roots on Rational Squares]
\label{thm:rational-square-root-exists-on-rational-squares}
If $a\in\mathbb{Q}$ is a rational square, then there exists
$x\in\mathbb{Q}$ such that
\[
x\ge 0_{\mathbb{Q}}
\quad\text{and}\quad
x^2=a.
\]
\hyperref[prf:rational-square-root-exists-on-rational-squares]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\forall a\in\mathbb{Q}\;
\bigl(\text{$a$ is a rational square}
\Rightarrow
\exists x\in\mathbb{Q}\,(x\ge 0_{\mathbb{Q}}\land x^2=a)\bigr).
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\text{$a$ is a rational square}
\Rightarrow
\exists x\in\mathbb{Q}\,(x\ge0_{\mathbb{Q}}\land x^2=a).
\]
\end{remark*}

\begin{remark*}[Interpretation]
Perfect rational squares have rational square roots; the nonnegative one is
obtained by taking the absolute value of any rational square root.
\end{remark*}

\begin{theorembox}{Theorem}
\begin{theorem}[Uniqueness of Nonnegative Rational Square Roots]
\label{thm:uniqueness-of-nonnegative-rational-square-roots}
If $x,y\in\mathbb{Q}$, $x\ge 0_{\mathbb{Q}}$, $y\ge 0_{\mathbb{Q}}$, and
\[
x^2=y^2,
\]
then $x=y$.
\hyperref[prf:uniqueness-of-nonnegative-rational-square-roots]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\forall x,y\in\mathbb{Q}\;
\bigl((x\ge 0_{\mathbb{Q}}\land y\ge 0_{\mathbb{Q}}\land x^2=y^2)
\Rightarrow x=y\bigr).
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{Nonnegative}(x)\land\operatorname{Nonnegative}(y)\land x^2=y^2
\Rightarrow x=y.
\]
\end{remark*}

\begin{remark*}[Interpretation]
The nonnegative sign convention makes a rational square root unique whenever
one exists.
\end{remark*}

\begin{definitionbox}{Definition}
\begin{definition}[Rational Square Root]
\label{def:rational-square-root}
For a rational square $a\in\mathbb{Q}$, define $\sqrt_{\mathbb{Q}}{a}$ to be
the unique rational number $x$ such that
\[
x\ge 0_{\mathbb{Q}}
\quad\text{and}\quad
x^2=a.
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\text{$a$ is a rational square}
\Rightarrow
\sqrt_{\mathbb{Q}}{a}
=\text{ the unique }x\in\mathbb{Q}\text{ with }x\ge0_{\mathbb{Q}}\text{ and }x^2=a.
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\sqrt_{\mathbb{Q}}{a}
=\iota x\,(x\in\mathbb{Q}\land x\ge0_{\mathbb{Q}}\land x^2=a).
\]
\end{remark*}

\begin{remark*}[Interpretation]
The rational square-root symbol is a partial operation: it is defined only on
rational numbers that are rational squares.
\end{remark*}

\begin{remark*}[Interpretation]
Existence supplies at least one nonnegative rational root on the stated
domain, and uniqueness makes the notation unambiguous.

\begin{theorembox}{Theorem}
\begin{theorem}[No Rational Square Root of Two]
\label{thm:no-rational-square-root-of-two}
There is no rational number $q\in\mathbb{Q}$ such that
\[
q\cdot q=2_{\mathbb{Q}}.
\]
\hyperref[prf:no-rational-square-root-of-two]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\neg\exists q\in\mathbb{Q}\,(q\cdot q=2_{\mathbb{Q}}).
\]
\end{remark*}

\begin{remark*}[Interpretation]
No rational number squares to rational two. Equivalently, the rational
square-root operation is not total on the nonnegative rational numbers.
\end{remark*}

\subsection*{The Square-Two Gap}

\begin{definitionbox}{Definition}
\begin{definition}[The Gap Set in $\mathbb{Q}$]
\label{def:gap-set-q}
Define
\[
S:=\{q\in\mathbb{Q}:0_{\mathbb{Q}}<q\ \text{and}\ q\cdot q<2_{\mathbb{Q}}\}.
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
S=\{q\in\mathbb{Q}:0_{\mathbb{Q}}<q\land q\cdot q<2_{\mathbb{Q}}\}.
\]
\end{remark*}

\begin{remark*}[Interpretation]
The gap set consists of positive rational numbers whose square lies below two.
\end{remark*}

\begin{propositionbox}{Proposition}
\begin{proposition}[The Gap Set Is Bounded Above in $\mathbb{Q}$]
\label{prop:gap-set-bounded-above-q}
The set $S$ is nonempty and bounded above in $\mathbb{Q}$.
\hyperref[prf:gap-set-bounded-above-q]{\textit{Go to proof.}}
\end{proposition}
\end{propositionbox}

\begin{remark*}[Standard quantified statement]
\[
S\ne\emptyset
\land
\exists b\in\mathbb{Q}\,\forall s\in S\,(s\le b).
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{BoundedAbove}(S,\mathbb{Q},\le)
\]
means that some rational number is an upper bound for every element of $S$.
\end{remark*}

\begin{remark*}[Interpretation]
The square-two gap set has rational elements and has rational upper bounds.
\end{remark*}

\begin{theorembox}{Theorem}
\begin{theorem}[The Square-Two Gap Has No Rational Supremum]
\label{thm:gap-set-no-rational-supremum-q}
The set $S$ has no least upper bound in $\mathbb{Q}$.
\hyperref[prf:gap-set-no-rational-supremum-q]{\textit{Go to proof.}}

\end{remark*}

\begin{remark*}[Interpretation]
Although $S$ is bounded above in $\mathbb{Q}$, no rational upper bound is least.

\begin{corollarybox}{Corollary}
\begin{corollary}[$\mathbb{Q}$ Is Order-Incomplete]
\label{cor:q-order-incomplete}
The ordered field $\mathbb{Q}$ is not order-complete: there exists a nonempty
bounded-above subset of $\mathbb{Q}$ with no supremum in $\mathbb{Q}$.
\hyperref[prf:q-order-incomplete]{\textit{Go to proof.}}
\end{corollary}
\end{corollarybox}

\[

\left(A\ne\emptyset
\land \exists b\in\mathbb{Q}\,\forall a\in A\,(a\le b)
\land \neg\exists u\in\mathbb{Q}\,\operatorname{LeastUpperBound}(u,A,\mathbb{Q},\le)\right).
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\neg\operatorname{OrderComplete}(\mathbb{Q},\le)
\]
means that a nonempty bounded-above subset of $\mathbb{Q}$ can fail to have a rational supremum.
\end{remark*}

\begin{remark*}[Interpretation]
The rational ordered field lacks the least-upper-bound property.
\end{remark*}

\subsection*{\texorpdfstring{Cauchy Sequences in $\mathbb{Q}$}{Cauchy Sequences in Q}}

\begin{definitionbox}{Definition}
\begin{definition}[Cauchy Sequence in $\mathbb{Q}$]
\label{def:cauchy-sequence-in-q}
A sequence $(x_n)$ in $\mathbb{Q}$ is Cauchy if
\[
\forall\varepsilon\in\mathbb{Q}\,
\left(
0_{\mathbb{Q}}<\varepsilon
\Longrightarrow
\exists N\in\mathbb{N}\,
\forall m,n\ge N\,
\left|x_m-x_n\right|<\varepsilon
\right).
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\operatorname{CauchySequence}_{\mathbb{Q}}((x_n))
\Longleftrightarrow
\forall\varepsilon\in\mathbb{Q}\,
\left(0_{\mathbb{Q}}<\varepsilon
\Longrightarrow
\exists N\in\mathbb{N}\,
\forall m,n\ge N\,(\left|x_m-x_n\right|<\varepsilon)\right).
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{CauchySequence}_{\mathbb{Q}}((x_n))
\]
means that the rational sequence $(x_n)$ is eventually pairwise arbitrarily close in $\mathbb{Q}$.
\end{remark*}

\begin{remark*}[Interpretation]
A rational Cauchy sequence has terms that eventually remain within any positive rational tolerance of one another.
\end{remark*}

\begin{lemmabox}{Lemma}
\begin{lemma}[Rational Limits Are Unique]
\label{lem:rational-limit-unique}
A convergent sequence in $\mathbb{Q}$ has at most one rational limit.
\hyperref[prf:rational-limit-unique]{\textit{Go to proof.}}
\end{lemma}
\end{lemmabox}

\begin{remark*}[Standard quantified statement]
\[
\forall (x_n)\,\forall a,b\in\mathbb{Q},\qquad
\left(\lim_{n\to\infty}x_n=a\land \lim_{n\to\infty}x_n=b\right)
\Longrightarrow a=b.
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{UniqueLimit}_{\mathbb{Q}}((x_n))
\]
means that any two rational limits of the sequence $(x_n)$ are equal.
\end{remark*}

\begin{remark*}[Interpretation]
A rational sequence cannot converge to two distinct rational limits.
\end{remark*}
