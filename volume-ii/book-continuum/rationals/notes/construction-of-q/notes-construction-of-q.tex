\begin{definition}[Nonzero Integers]
\label{def:nonzero-integers}
The set of nonzero integers is
\[
\mathbb{Z}^{\ast}:=\mathbb{Z}\setminus\{0_{\mathbb{Z}}\}.
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[

\operatorname{NonzeroCarrier}(\mathbb{Z}^{\ast},\mathbb{Z},0_{\mathbb{Z}}).
\]
\end{remark*}

\begin{remark*}[Interpretation]
The nonzero integers form the allowed denominator set for rational
prefractions.
\end{remark*}

\begin{definitionbox}{Definition}
\begin{definition}[Prefraction]
\label{def:prefraction}
A \emph{prefraction} is an ordered pair $(a,b)\in\mathbb{Z}\times\mathbb{Z}^{\ast}$,
read as the formal quotient $a/b$.
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
(a,b)\in\mathbb{Z}\times\mathbb{Z}^{\ast}.
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{Prefraction}((a,b)).
\]
\end{remark*}

\begin{remark*}[Interpretation]
A prefraction is the raw ordered-pair datum used to model a formal quotient.
\end{remark*}

\subsection*{Fraction Equivalence}

\begin{definitionbox}{Definition}
\begin{definition}[Fraction Equivalence]
\label{def:fraction-equivalence}
For $(a,b),(c,d)\in\operatorname{Pre}\mathbb{Q}$, define
\[
(a,b)\sim_{\mathbb{Q}}(c,d)
\quad:\Longleftrightarrow\quad
a\cdot d=b\cdot c.
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\forall (a,b),(c,d)\in\operatorname{Pre}\mathbb{Q},\quad
(a,b)\sim_{\mathbb{Q}}(c,d)
\Longleftrightarrow
a\cdot d=b\cdot c.
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{FractionEquivalent}((a,b),(c,d)).
\]
\end{remark*}

\begin{remark*}[Interpretation]
Two prefractions represent the same quotient exactly when cross-multiplication
balances them.
\end{remark*}
