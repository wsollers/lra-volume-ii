\subsection*{\texorpdfstring{Derived Field Laws on $\mathbb{Q}$}{Derived Field Laws on Q}}

\begin{corollarybox}{Corollary}
\begin{corollary}[Zero Product on $\mathbb{Q}$]
\label{cor:zero-product-on-q}
For all $x,y\in\mathbb{Q}$,
\[
x\cdot y=0_{\mathbb{Q}}
\quad\Longleftrightarrow\quad
x=0_{\mathbb{Q}}\ \text{or}\ y=0_{\mathbb{Q}}.
\]
\hyperref[prf:zero-product-on-q]{\textit{Go to proof.}}
\end{corollary}
\end{corollarybox}

\begin{remark*}[Standard quantified statement]
For all $x,y\in\mathbb{Q}$,
\[
x\cdot y=0_{\mathbb{Q}}
\Longleftrightarrow
x=0_{\mathbb{Q}}\lor y=0_{\mathbb{Q}}.
\]
\end{remark*}

\begin{remark*}[Interpretation]
The rational field has no nonzero zero divisors.
\end{remark*}

\subsection*{\texorpdfstring{Integer Powers on $\mathbb{Q}$}{Integer Powers on Q}}

\begin{definitionbox}{Definition}
\begin{definition}[Integer Power on $\mathbb{Q}$]
\label{def:integer-power-on-q}
Let $x\in\mathbb{Q}$. For $n\in\mathbb{W}$, define $x^n$ by the usual
recursive power operation. If $x\ne 0_{\mathbb{Q}}$ and $n\in\mathbb{Z}$ is
negative, define
\[
x^n:=(x^{-1})^{-n}.
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
For $x\in\mathbb{Q}$ and $n\in\mathbb{Z}$, $x^n$ is the recursive nonnegative
power when $n\in\mathbb{W}$, and if $x\ne0_{\mathbb{Q}}$ and $n<0$, then
\[
x^n=(x^{-1})^{-n}.
\]
\end{remark*}

\begin{remark*}[Interpretation]
Integer powers of nonzero rationals extend whole-number powers by using
reciprocals for negative exponents.
\end{remark*}

\begin{corollary}[Exponent Addition Law on $\mathbb{Q}$]
\label{cor:exponent-addition-on-q}
For every nonzero $x\in\mathbb{Q}$ and all $m,n\in\mathbb{Z}$,
\[
x^{m+n}=x^m\cdot x^n.
\]
\hyperref[prf:exponent-addition-on-q]{\textit{Go to proof.}}

$m,n\in\mathbb{Z}$,
\[
x^{m+n}=x^m\cdot x^n.
\]
\end{remark*}

\begin{remark*}[Interpretation]
For nonzero rationals, adding exponents corresponds to multiplying powers.
\end{remark*}
