\subsection*{\texorpdfstring{Addition on $\mathbb{Q}$-Classes}{Addition on Q-Classes}}

\begin{definitionbox}{Definition}
\begin{definition}[Addition on $\mathbb{Q}$-Classes]
\label{def:addition-on-q-classes}
Define addition on representatives by
\[
[a,b]+[c,d]:=[a\cdot d+b\cdot c,\ b\cdot d].
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
For all $(a,b),(c,d)\in\operatorname{Pre}\mathbb{Q}$,
\[
[a,b]+[c,d]=[a\cdot d+b\cdot c,\ b\cdot d].
\]
\end{remark*}

\begin{remark*}[Interpretation]
Rational addition is first defined on prefraction representatives by the
usual cross-multiplication formula. The subsequent well-definedness theorem
shows that this representative formula descends to the quotient
$\mathbb{Q}$.
\end{remark*}

\begin{lemmabox}{Lemma}
\begin{lemma}[Addition Lands in $\operatorname{Pre}\mathbb{Q}$]
\label{lem:addition-denominator-nonzero-q}
If $b\ne 0_{\mathbb{Z}}$ and $d\ne 0_{\mathbb{Z}}$, then
\[
b\cdot d\ne 0_{\mathbb{Z}}.
\]
\hyperref[prf:addition-denominator-nonzero-q]{\textit{Go to proof.}}
\end{lemma}
\end{lemmabox}

\begin{remark*}[Standard quantified statement]
For all $(a,b),(c,d)\in\operatorname{Pre}\mathbb{Q}$,
\[
b\ne 0_{\mathbb{Z}}\land d\ne 0_{\mathbb{Z}}
\Longrightarrow b\cdot d\ne 0_{\mathbb{Z}}.
\]
\end{remark*}

\begin{remark*}[Interpretation]
The denominator in the representative formula for rational addition is a
nonzero integer, so the displayed sum is again a prefraction.
\end{remark*}

\begin{lemmabox}{Lemma}
\begin{lemma}[Addition Respects Fraction Equivalence on the Left]
\label{lem:addition-on-q-respects-left-equivalence}
If $[a,b]=[a',b']$, then
\[
[a,b]+[c,d]=[a',b']+[c,d].
\]
\hyperref[prf:addition-on-q-respects-left-equivalence]{\textit{Go to proof.}}
\end{lemma}
\end{lemmabox}

\begin{remark*}[Standard quantified statement]
For all prefractions $(a,b),(a',b'),(c,d)$,
\[
[a,b]=[a',b']
\Longrightarrow
[a,b]+[c,d]=[a',b']+[c,d].
\]
\end{remark*}

\begin{remark*}[Interpretation]
Changing the left representative within its fraction-equivalence class does
not change the rational sum.
\end{remark*}
