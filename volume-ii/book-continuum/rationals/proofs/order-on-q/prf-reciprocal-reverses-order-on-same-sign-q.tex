\begin{theorem*}[Reciprocal Reverses Order on Same-Sign Rationals]
For all nonzero $x,y\in\mathbb{Q}$,
\[
0_{\mathbb{Q}}<x<y
\Longrightarrow
0_{\mathbb{Q}}<y^{-1}<x^{-1},
\qquad
x<y<0_{\mathbb{Q}}
\Longrightarrow
y^{-1}<x^{-1}<0_{\mathbb{Q}}.
\]
\end{theorem*}
