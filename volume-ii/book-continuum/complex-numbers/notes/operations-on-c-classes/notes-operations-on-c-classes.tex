\subsection*{\texorpdfstring{Addition on $\mathbb{C}$-Classes}{Addition on C-Classes}}

\begin{definitionbox}{Definition}
\begin{definition}[Addition on $\mathbb{C}$-Classes]
\label{def:addition-on-c-classes}
Define addition on representatives by
\[
[a,b]_{\mathbb{C}}+[c,d]_{\mathbb{C}}
:=[a+c,b+d]_{\mathbb{C}}.
\]
\end{definition}
\end{definitionbox}

\LeanFormalizes{def:addition-on-c-classes}{lra-lean}{LRA.VolumeII.ComplexNumbers.Construction.Construction}{addition}{pending}

\begin{remark*}[Standard quantified statement]
For all real coordinates $a,b,c,d\in\mathbb{R}$,
\[
[a,b]_{\mathbb{C}}+[c,d]_{\mathbb{C}}
=[a+c,b+d]_{\mathbb{C}}.
\]
\end{remark*}

\begin{remark*}[Interpretation]
Complex addition is defined coordinatewise on representatives.
\end{remark*}

\begin{lemmabox}{Lemma}
\begin{lemma}[Addition Respects Complex Coordinate Equivalence]
\label{lem:addition-on-c-respects-equivalence}
If $[a,b]_{\mathbb{C}}=[a',b']_{\mathbb{C}}$ and
$[c,d]_{\mathbb{C}}=[c',d']_{\mathbb{C}}$, then
\[
[a,b]_{\mathbb{C}}+[c,d]_{\mathbb{C}}
=
[a',b']_{\mathbb{C}}+[c',d']_{\mathbb{C}}.
\]
\hyperref[prf:addition-on-c-respects-equivalence]{\textit{Go to proof.}}
\end{lemma}
\end{lemmabox}

\begin{remark*}[Standard quantified statement]
For all real coordinates $a,b,a',b',c,d,c',d'\in\mathbb{R}$,
\[
[a,b]_{\mathbb{C}}=[a',b']_{\mathbb{C}}
\land
[c,d]_{\mathbb{C}}=[c',d']_{\mathbb{C}}
\Longrightarrow
[a,b]_{\mathbb{C}}+[c,d]_{\mathbb{C}}
=
[a',b']_{\mathbb{C}}+[c',d']_{\mathbb{C}}.
\]
\end{remark*}

\begin{remark*}[Interpretation]
Complex addition is independent of the chosen coordinate representatives.
\end{remark*}

  \item \hyperref[def:complex-coordinate-equivalence]{Complex Coordinate Equivalence}
\end{itemize}
\end{dependencies}

\begin{theorembox}{Theorem}
\begin{theorem}[Addition on $\mathbb{C}$ Is Well-Defined]
\label{thm:addition-on-c-well-defined}
Addition determines a well-defined binary operation
\[
\mathbb{C}\times\mathbb{C}\to\mathbb{C}.
\]
\hyperref[prf:addition-on-c-well-defined]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\operatorname{WellDefinedBinaryOperation}
\bigl(+, \operatorname{Quotient}(\mathbb{R}\times\mathbb{R},\sim_{\mathbb{C}})\bigr).
\]
\end{remark*}

\begin{remark*}[Interpretation]
Suppose the input classes are represented in two ways. Coordinate equivalence
gives \(a=a'\), \(b=b'\), 
\(c=c'\), and \(d=d'\). Addition in
\(\mathbb{R}\) gives \(a+c=a'+c'\) and \(b+d=b'+d'\), so the resulting
complex classes agree. Thus addition descends to a binary operation on
$\mathbb{C}$.
\end{remark*}

-[a,b]_{\mathbb{C}}:=[-a,-b]_{\mathbb{C}}.

\subsection*{\texorpdfstring{Multiplication on $\mathbb{C}$-Classes}{Multiplication on C-Classes}}

\begin{definitionbox}{Definition}
\begin{definition}[Multiplication on $\mathbb{C}$-Classes]
\label{def:multiplication-on-c-classes}
Define multiplication on representatives by
\[
[a,b]_{\mathbb{C}}\cdot[c,d]_{\mathbb{C}}
:=[ac-bd,ad+bc]_{\mathbb{C}}.
\]
\end{definition}
\end{definitionbox}

\LeanFormalizes{def:multiplication-on-c-classes}{lra-lean}{LRA.VolumeII.ComplexNumbers.Construction.Construction}{multiplication}{pending}

\begin{remark*}[Standard quantified statement]
For all $a,b,c,d\in\mathbb{R}$,
\[
[a,b]_{\mathbb{C}}\cdot[c,d]_{\mathbb{C}}
=[ac-bd,ad+bc]_{\mathbb{C}}.
\]
\end{remark*}

\begin{remark*}[Interpretation]
The rule is forced by the intended identity \(i^2=-1\):
\[
(a+bi)(c+di)=(ac-bd)+(ad+bc)i.
\]
\end{remark*}

\subsection*{Conjugation and Inverses}

\begin{definitionbox}{Definition}
\begin{definition}[Complex Conjugation]
\label{def:complex-conjugation}
For \(z=[a,b]_{\mathbb{C}}\), define its \emph{complex conjugate} by
\[
\overline{z}:=[a,-b]_{\mathbb{C}}.
\]
\end{definition}
\end{definitionbox}

\LeanFormalizes{def:complex-conjugation}{lra-lean}{LRA.VolumeII.ComplexNumbers.Construction.Construction}{conjugation}{pending}

\begin{remark*}[Standard quantified statement]
For every $z=[a,b]_{\mathbb{C}}\in\mathbb{C}$,
\[
\overline{z}=[a,-b]_{\mathbb{C}}.
\]
\end{remark*}

\begin{remark*}[Interpretation]
Complex conjugation keeps the real coordinate and negates the imaginary
coordinate.
\end{remark*}

  \item \hyperref[def:complex-class-notation]{Complex Class Notation}

\begin{theorembox}{Theorem}
\begin{theorem}[Complex Conjugation Is Well-Defined]
\label{thm:complex-conjugation-well-defined}
Complex conjugation determines a well-defined unary operation
\(\mathbb{C}\to\mathbb{C}\).
\hyperref[prf:complex-conjugation-well-defined]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\operatorname{WellDefinedUnaryOperation}
\bigl(\overline{(\cdot)},\operatorname{Quotient}(\mathbb{R}\times\mathbb{R},\sim_{\mathbb{C}})\bigr).
\]
\end{remark*}

\begin{remark*}[Interpretation]
Equivalent prepairs have equal first coordinates and equal second coordinates.
Keeping the first coordinate and negating the second preserves that equality,
so conjugation does not depend on the representative.
\end{remark*}

\begin{lemmabox}{Lemma}
\begin{lemma}[Product with the Con
jugate Is Real and Nonnegative]
\label{lem:complex-product-with-conjugate}
For \(z=[a,b]_{\mathbb{C}}\),
\[
z\overline{z}=[a^2+b^2,0]_{\mathbb{C}}.
\]
Moreover, if \(z\ne 0_{\mathbb{C}}\), then \(a^2+b^2\ne 0\).
\hyperref[prf:complex-product-with-conjugate]{\textit{Go to proof.}}
\end{lemma}
\end{lemmabox}

\LeanFormalizes{lem:complex-product-with-conjugate}{lra-lean}{LRA.VolumeII.ComplexNumbers.Construction.Construction}{norm_squared}{pending}

\begin{remark*}[Standard quantified statement]
For every $z=[a,b]_{\mathbb{C}}\in\mathbb{C}$,
\[
z\overline{z}=[a^2+b^2,0]_{\mathbb{C}},
\qquad
z\ne 0_{\mathbb{C}}\Longrightarrow a^2+b^2\ne 0.
\]
\end{remark*}

\begin{remark*}[Interpretation]
Multiplying a complex number by its conjugate yields a real nonnegative
coordinate, and this coordinate is nonzero for nonzero complex numbers.
\end{remark*}

\begin{definitionbox}{Definition}
\begin{definition}[Multiplicative Inverse in $\mathbb{C}$]
\label{def:multiplicative-inverse-on-c}
For \(z=[a,b]_{\mathbb{C}}\ne 0_{\mathbb{C}}\), define
\[
z^{-1}
:=
\left[\frac{a}{a^2+b^2},\frac{-b}{a^2+b^2}\right]_{\mathbb{C}}.
\]
\end{definition}
\end{definitionbox}

\LeanFormalizes{def:multiplicative-inverse-on-c}{lra-lean}{LRA.VolumeII.ComplexNumbers.Construction.Construction}{inverse}{pending}

\begin{remark*}[Standard quantified statement]
For every $z=[a,b]_{\mathbb{C}}\in\mathbb{C}\setminus\{0_{\mathbb{C}}\}$,
\[
z^{-1}
=
\left[\frac{a}{a^2+b^2},\frac{-b}{a^2+b^2}\right]_{\mathbb{C}}.
\]
\end{remark*}

\begin{remark*}[Interpretation]
The inverse of a nonzero complex number is obtained by dividing its conjugate
coordinates by $a^2+b^2$.
\end{remark*}
