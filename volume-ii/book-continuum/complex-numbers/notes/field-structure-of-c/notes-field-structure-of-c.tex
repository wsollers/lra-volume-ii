\subsection*{The Imaginary Unit}

\begin{theorembox}{Theorem}
\begin{theorem}[The Imaginary Unit Squares to Negative One]
\label{thm:i-squared-negative-one}
In \(\mathbb{C}\),
\[
i^2=-1_{\mathbb{C}}.
\]
\hyperref[prf:i-squared-negative-one]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
i^2=-1_{\mathbb{C}}.
\]
\end{remark*}

\begin{remark*}[Interpretation]
The imaginary unit is a complex number whose square is negative one.
\end{remark*}

\begin{remark*}[Interpretation]
The equation \(i^2=-1\) is the structural reason for introducing
\(\mathbb{C}\). It makes the polynomial \(x^2+1\) have a root, while the field
operations remain compatible with the real arithmetic already constructed.
\end{remark*}

\begin{theorembox}{Theorem}
\begin{theorem}[$\mathbb{C}$ Is Not an Ordered Field]
\label{thm:c-is-not-an-ordered-field}
There is no order relation on \(\mathbb{C}\) making \(\mathbb{C}\) an ordered
field and extending the usual order on \(\mathbb{R}\).
\hyperref[prf:c-is-not-an-ordered-field]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
There is no relation $<_{\mathbb{C}}$ such that
\[
\operatorname{IsOrderedField}
\bigl(\operatorname{OrderedField}(
\operatorname{Field}(
\operatorname{CommutativeRing}(\operatorname{Ring}(
\mathbb{C},+,\cdot,0_{\mathbb{C}},1_{\mathbb{C}}))),
<_{\mathbb{C}})\bigr)
\]
and $<_{\mathbb{C}}$ extends the usual order on $\mathbb{R}$.
\end{remark*}

\begin{remark*}[Interpretation]
The complex field cannot be equipped with an ordered-field order compatible
with the real order.
\end{remark*}

\begin{remark*}[Interpretation]
In any ordered field,
every square is nonnegative. If \(\mathbb{C}\) were an
ordered field extending the real order, \(i^2\) would be nonnegative. But
\(i^2=-1\), and \(-1<0\) in the real subfield.
\end{remark*}
