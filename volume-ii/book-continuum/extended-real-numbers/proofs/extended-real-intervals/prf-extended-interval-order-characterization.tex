\begin{theorem*}[Extended Interval Order Characterization]
A subset $I\subseteq\overline{\mathbb{R}}$ is an extended real interval if and
only if for all $x,z\in I$ with $x\le z$,
\[
[x,z]_{\overline{\mathbb{R}}}\subseteq I.
\]
\end{theorem*}
