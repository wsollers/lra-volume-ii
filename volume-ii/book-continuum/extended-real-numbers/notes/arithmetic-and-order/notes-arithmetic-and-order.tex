\begin{definitionbox}{Definition (Domain of Extended Real Addition)}
\begin{definition}[Domain of Extended Real Addition]
\label{def:extended-real-addition-domain}
Extended addition is defined for pairs $(x,y)\in\overline{\mathbb{R}}^2$
except for the two indeterminate pairs
\[
(+\infty,-\infty)
\quad\text{and}\quad
(-\infty,+\infty).
\]
\end{definition}
\end{definitionbox}
\begin{remark*}[Standard quantified statement]
\[
x+y\text{ is defined}
\Longleftrightarrow
\neg\bigl((x,y)=(+\infty,-\infty)\lor(x,y)=(-\infty,+\infty)\bigr).
\]
\end{remark*}
\begin{remark*}[Interpretation]
Adding opposite infinities is deliberately left undefined.
\end{remark*}
\begin{dependencies}
\hyperref[def:extended-real-numbers]{Extended Real Numbers}.
\end{dependencies}

\begin{definitionbox}{Definition (Extended Real Addition)}
\begin{definition}[Extended Real Addition]
\label{def:extended-real-addition}
If $x+y$ is defined in $\overline{\mathbb{R}}$, then it agrees with ordinary
real addition when $x,y\in\mathbb{R}$, and satisfies
\[
x+(+\infty)=+\infty,\qquad x+(-\infty)=-\infty
\]
whenever these expressions are defined.
\end{definition}
\end{definitionbox}
\begin{remark*}[Standard quantified statement]
\[
x,y\in\mathbb{R}\Rightarrow
x+_{\overline{\mathbb{R}}}y=x+_{\mathbb{R}}y.
\]
\end{remark*}
\begin{remark*}[Interpretation]
Finite translations do not move an infinite endpoint away from infinity.
\end{remark*}

\begin{theorembox}{Theorem (Extended Real Addition Is Well Defined)}
\begin{theorem}[Extended Real Addition Is Well Defined]
\label{thm:extended-real-addition-well-defined}
For every pair $(x,y)$ in the domain of extended real addition, there is a
unique element $z\in\overline{\mathbb{R}}$ such that
\[
z=x+y.
\]
\hyperref[prf:extended-real-addition-well-defined]{Go to proof.}
\end{theorem}
\end{theorembox}
\begin{remark*}[Standard quantified statement]
\[
\forall x,y\in\overline{\mathbb{R}}\;
\bigl((x,y)\in\operatorname{Dom}(+)\Rightarrow
\exists!z\in\overline{\mathbb{R}}\;(z=x+y)\bigr).
\]
\end{remark*}
\begin{remark*}[Predicate reading]
\[
\operatorname{WellDefined}(+,\operatorname{Dom}(+),\overline{\mathbb{R}})
\]
means that every allowed pair of extended real inputs has a unique extended real sum.
\end{remark*}
\begin{remark*}[Negated quantified statement]
\[
\exists x,y\in\overline{\mathbb{R}}\;
\bigl((x,y)\in\operatorname{Dom}(+)\land
\neg\exists!z\in\overline{\mathbb{R}}\;(z=x+y)\bigr).
\]
\end{remark*}

\begin{remark*}[Negation predicate reading]
\[
\exists x,y\in\overline{\mathbb{R}}\;
\bigl((x,y)\in\operatorname{Dom}(+)\land
\neg\exists!z\in\overline{\mathbb{R}}\;(z=x+y)\bigr).
\]
\end{remark*}

\begin{remark*}[Failure modes]
\begin{description}
  \item[Exposition.] The named branch below records the displayed negation of the predicate.
  \item[\(\operatorname{WellDefined}\) fails.] The predicate fails exactly in the displayed negation case:
  \[
  \exists x,y\in\overline{\mathbb{R}}\;
  \bigl((x,y)\in\operatorname{Dom}(+)\land
  \neg\exists!z\in\overline{\mathbb{R}}\;(z=x+y)\bigr).
  \]
  In predicate-reading form this is recorded as
  \[
  \exists x,y\in\overline{\mathbb{R}}\;
  \bigl((x,y)\in\operatorname{Dom}(+)\land
  \neg\exists!z\in\overline{\mathbb{R}}\;(z=x+y)\bigr).
  \]
\end{description}
\end{remark*}
\begin{remark*}[Interpretation]
Once the two opposite-infinity cases are excluded, addition has a single
extended real value.
\end{remark*}

\hyperref[def:extended-real-addition]{Extended Real Addition}.
\end{dependencies}

\begin{theorembox}{Theorem (Extended Addition Extends Real Addition)}
\begin{theorem}[Extended Addition Extends Real Addition]
\label{thm:extended-real-addition-extends-real-addition}
For all $x,y\in\mathbb{R}$,
\[
x+_{\overline{\mathbb{R}}}y=x+_{\mathbb{R}}y.
\]
\hyperref[prf:extended-real-addition-extends-real-addition]{Go to proof.}
\end{theorem}
\end{theorembox}
\begin{remark*}[Standard quantified statement]
\[
\forall x,y\in\mathbb{R}\;(x+_{\overline{\mathbb{R}}}y=x+_{\mathbb{R}}y).
\]
\end{remark*}
\begin{remark*}[Predicate reading]
\[
\operatorname{Extends}(+_{\overline{\mathbb{R}}},+_{\mathbb{R}},\mathbb{R})
\]
means that extended addition agrees with real addition on finite real inputs.
\end{remark*}
\begin{remark*}[Interpretation]
No finite arithmetic is changed by passing to the extended reals.
\end{remark*}

\begin{theorembox}{Theorem (Addition with Infinities)}
\begin{theorem}[Addition with Infinities]
\label{thm:extended-real-addition-with-infinities}
For every $x\in\mathbb{R}$,
\[
x+(+\infty)=+\infty
\quad\text{and}\quad
x+(-\infty)=-\infty.
\]
Moreover,
\[
(+\infty)+(+\infty)=+\infty
\quad\text{and}\quad
(-\infty)+(-\infty)=-\infty.
\]
\hyperref[prf:extended-real-addition-with-infinities]{Go to proof.}
\end{theorem}
\end{theorembox}
\begin{remark*}[Standard quantified statement]
\[
\forall x\in\mathbb{R}\;(x+(+\infty)=+\infty\land x+(-\infty)=-\infty).
\]
\end{remark*}
\begin{remark*}[Interpretation]
Adding a finite real cannot cancel or change an infinite endpoint.
\end{remark*}

\hyperref[def:extended-real-addition]{Extended Real Addition}.
\end{dependencies}

\begin{definitionbox}{Definition (Extended Real Negation and Subtraction)}
\begin{definition}[Extended Real Negation and Subtraction]
\label{def:extended-real-negation}
Negation on $\overline{\mathbb{R}}$ extends ordinary real negation and is
defined by
\[
-(+\infty)=-\infty,\qquad -(-\infty)=+\infty.
\]
Subtraction is defined by
\[
x-y=x+(-y)
\]
whenever the right-hand side is defined.
\end{definition}
\end{definitionbox}
\begin{remark*}[Standard quantified statement]
\[
x-y\text{ is defined}\Longleftrightarrow x+(-y)\text{ is defined}.
\]
\end{remark*}
\begin{remark*}[Interpretation]
Subtraction inherits the same opposite-infinity exclusions from addition.
\end{remark*}
\begin{dependencies}
\hyperref[def:extended-real-addition]{Extended Real Addition};
\hyperref[def:negation-on-r]{Negation of a Cut}.
\end{dependencies}

\begin{definitionbox}{Definition (Domain of Extended Real Multiplication)}
\begin{definition}[Domain of Extended Real Multiplication]
\label{def:extended-real-multiplication-domain}
Extended multiplication is defined for pairs $(x,y)\in\overline{\mathbb{R}}^2$
except when one factor is $0$ and the other factor is infinite.
\end{definition}
\end{definitionbox}
\begin{remark*}[Standard quantified statement]
\[
xy\text{ is undefined}
\Longleftrightarrow
(x=0\land y\in\{-\infty,+\infty\})\lor
(y=0\land x\in\{-\infty,+\infty\}).
\]
\end{remark*}
\begin{remark*}[Interpretation]
The forms $0\cdot(+\infty)$ and $0\cdot(-\infty)$ are indeterminate.
\end{remark*}
\begin{dependencies}
\hyperref[def:extended-real-numbers]{Extended Real Numbers}.
\end{dependencies}

\begin{definitionbox}{Definition (Extended Real Multiplication)}
\begin{definition}[Extended Real Multiplication]
\label{def:extended-real-multiplication}
Extended multiplication agrees with ordinary real multiplication on finite
inputs. If $x\in\mathbb{R}$ is positive, then
\[
x(+\infty)=+\infty,\qquad x(-\infty)=-\infty,
\]
and if $x\in\mathbb{R}$ is negative, then
\[
x(+\infty)=-\infty,\qquad x(-\infty)=+\infty.
\]
Products of two infinite endpoints follow the corresponding sign rule.
\end{definition}
\end{definitionbox}
\begin{remark*}[Standard quantified statement]
\[
x,y\in\mathbb{R}\Rightarrow
x\cdot_{\overline{\mathbb{R}}}y=x\cdot_{\mathbb{R}}y.
\]
\end{remark*}
\begin{remark*}[Interpretation]
Multiplication by infinity records sign and magnitude separately.
\end{remark*}

\hyperref[def:extended-real-multiplication-domain]{Domain of Extended Real Multiplication};
\hyperref[def:multiplication-on-r]{Multiplication of Cuts}.

\begin{theorembox}{Theorem (Multiplication with Infinities)}
\begin{theorem}[Multiplication with Infinities]
\label{thm:extended-real-multiplication-with-infinities}
If $x\in\mathbb{R}$ and $x>0$, then
\[
x(+\infty)=+\infty
\quad\text{and}\quad
x(-\infty)=-\infty.
\]
If $x\in\mathbb{R}$ and $x<0$, then
\[
x(+\infty)=-\infty
\quad\text{and}\quad
x(-\infty)=+\infty.
\]
\hyperref[prf:extended-real-multiplication-with-infinities]{Go to proof.}
\end{theorem}
\end{theorembox}
\begin{remark*}[Standard quantified statement]
\[
\forall x\in\mathbb{R}\;
\bigl(x>0\Rightarrow x(+\infty)=+\infty\bigr)
\land
\bigl(x<0\Rightarrow x(+\infty)=-\infty\bigr).
\]
\end{remark*}
\begin{remark*}[Interpretation]
Multiplication by a positive finite number preserves the direction of
infinity; multiplication by a negative finite number reverses it.
\end{remark*}

\hyperref[def:extended-real-multiplication]{Extended Real Multiplication};
\hyperref[def:strict-order-on-r]{Strict Order on $\mathbb{R}$}.
\end{dependencies}

\begin{propositionbox}{Proposition (Indeterminate Extended Real Forms)}
\begin{proposition}[Indeterminate Extended Real Forms]
\label{prop:indeterminate-extended-real-forms}
The expressions
\[
(+\infty)+(-\infty),\qquad
(-\infty)+(+\infty),\qquad
0\cdot(+\infty),\qquad
0\cdot(-\infty)
\]
are not defined in the extended real arithmetic introduced here.
\hyperref[prf:indeterminate-extended-real-forms]{Go to proof.}
\end{proposition}
\end{propositionbox}
\begin{remark*}[Standard quantified statement]
\[
\{(+\infty,-\infty),(-\infty,+\infty)\}\cap\operatorname{Dom}(+)=\varnothing
\]
and
\[
\{(0,+\infty),(0,-\infty),(+\infty,0),(-\infty,0)\}
\cap\operatorname{Dom}(\cdot)=\varnothing.
\]
\end{remark*}
\begin{remark*}[Interpretation]
The extended reals are useful because these ambiguous forms are excluded, not
because every formal expression is assigned a value.
\end{remark*}

\hyperref[def:extended-real-addition-domain]{Domain of Extended Real Addition};
\hyperref[def:extended-real-multiplication-domain]{Domain of Extended Real Multiplication}.
\end{dependencies}
