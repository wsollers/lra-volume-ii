\section{\texorpdfstring{Axiom Systems for $\mathbb{N}$, $\mathbb{W}$, $\mathbb{Z}$, $\mathbb{Q}$, and $\mathbb{R}$}{Axiom Systems for N, W, Z, Q, and R}}
\label{sec:number-systems-axioms}

\begin{toolkitbox}{Derived Properties --- Quick Reference}
\small
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{@{} l p{0.68\linewidth} @{}}
\toprule
\textbf{Name} & \textbf{Meaning} \\
\midrule
Zero multiplication
  & $a \cdot 0 = 0$; prove from
    $a\cdot 0 = a\cdot(0+0) = a\cdot 0 + a\cdot 0$ and cancel
    $a\cdot 0$ via A4. \\
Negation rule
  & $(-a)\cdot b = -(ab)$; use
    $ab + (-a)b = (a + (-a))b = 0\cdot b = 0$. \\
$(-1)$ rule
  & $(-1)\cdot a = -a$; this is the negation rule with $b = a$. \\
Double negation
  & $(-a)(-b) = ab$; compute
    $(-a)(-b) = -(a(-b)) = -(-(ab)) = ab$. \\
No zero divisors
  & $ab = 0 \Rightarrow a=0$ or $b=0$; for $\mathbb{Z}$ this follows
    from the integral-domain argument using well-ordering and signs. \\
\bottomrule
\end{tabular}
\end{toolkitbox}
