\begin{lemma}[Totality of the Minimal Iterator Relation]
\label{lem:minimal-iterator-relation-complete}
Let $(P,S,1)$ be a Peano system, let $(W,c,g)$ be iterator data for it, and let
$R^*$ be the minimal iterator relation. For each $n\in P$, there exists
$w\in W$ such that $(n,w)\in R^*$.
\hyperref[prf:minimal-iterator-relation-complete]{Go to proof.}
\end{lemma}
\end{lemmabox}

\begin{leancode}
theorem minimal_iterator_relation_complete
    (ps : PeanoSystem)
    (data : IteratorData ps) :
    forall element : ps.carrier,
      Exists (fun value : data.target =>
        minimal_iterator_relation ps data element value) := by
  apply induction_principle ps

  case base_case =>
    exact
      Exists.intro
        data.initial_value
        (minimal_iterator_relation_is_iterator_relation ps data).left

  case successor_step =>
    intro element induction_hypothesis
    cases induction_hypothesis with
    | intro value value_is_forced =>
        exact
          Exists.intro
            (data.step_rule value)
            ((minimal_iterator_relation_is_iterator_relation ps data).right
              element value value_is_forced)
\end{leancode}
\end{formalizationrecord}

\begin{remark*}[Standard quantified statement]
\[
\forall n\in P\;\exists w\in W\;((n,w)\in R^*).
\]
\end{remark*}

\begin{remark*}[Predicate reading]

\operatorname{TotalOnDomain}(R^*,P,W).
\]
\end{remark*}

\begin{remark*}[Negated quantified statement]
\[
\exists n\in P\;\forall w\in W\;((n,w)\notin R^*).
\]
\end{remark*}

\begin{remark*}[Failure modes]
\begin{description}
  \item[Exposition.]
  Totality fails exactly when some stage of $P$ receives no value in the
  minimal iterator relation.
  \item[Missing value.]
  \[
  \neg\operatorname{TotalOnDomain}(R^*,P,W).
  \]
  \[
  \exists n\in P\;\forall w\in W\;((n,w)\notin R^*).
  \]
\end{description}
\end{remark*}

\begin{remark*}[Interpretation]
This is the totality half of Feferman's minimal-relation construction for
recursive definition: after determinacy gives at most one value at each stage,
totality gives at least one value at each stage
\citep{FefermanNumberSystems1964}.
\end{remark*}
