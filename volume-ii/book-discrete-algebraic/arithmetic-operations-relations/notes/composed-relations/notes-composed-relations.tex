\begin{topicbox}{Definition (Preorder)}
\begin{exposition}
A \emph{preorder} is a reflexive and transitive relation.
\end{exposition}
\end{topicbox}

\begin{topicbox}{Definition (Partial Order)}
\begin{exposition}
A \emph{partial order} is a preorder that is antisymmetric.
\end{exposition}
\end{topicbox}

\begin{definitionbox}{Definition (Strict Partial Order)}
\begin{definition}[Strict Partial Order]
\label{def:strict-partial-order}
A \emph{strict partial order} is an irreflexive and transitive relation.
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\operatorname{Irreflexive}(R,A)
\quad\text{and}\quad
\operatorname{Transitive}(R,A).
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{StrictPartialOrder}(R,A).
\]
\end{remark*}

\begin{remark*}[Interpretation]
A strict partial order is a relation with no self-comparisons and transitive
comparison chains.
\end{remark*}

\begin{topicbox}{Definition (Total Order)}
\begin{exposition}
A \emph{total order} is a partial order whose relation is total.
\end{exposition}
\end{topicbox}

\begin{definitionbox}{Definition (Strict Total Order)}
\begin{definition}[Strict Total Order]
\label{def:strict-total-order}
A \emph{strict total order} is a strict partial order satisfying trichotomy.
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\operatorname{StrictPartialOrder}(<,A)
\quad\text{and}\quad
\operatorname{Trichotomous}(<,A).
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{StrictTotalOrder}(<,A).
\]
\end{remark*}

\begin{remark*}[Interpretation]
A strict total order is a strict partial order in which every pair is
trichotomously comparable.
\end{remark*}

\begin{dependencies}
\begin{itemize}
  \item \hyperref[def:strict-partial-order]{Strict Partial Order}
  \item \hyperref[def:trichotomy]{Trichotomy}
\end{itemize}
\end{dependencies}

\begin{topicbox}{Definition (Well-Order)}
\begin{exposition}
A \emph{well-order} is a total order in which every nonempty subset has a
least element.
\end{exposition}
\end{topicbox}
