\begin{propositionbox}{Proposition (Left Zero Absorption)}
\begin{proposition}[Left Zero Absorption]
\label{prop:left-zero-absorption}
When multiplication distributes over addition and $0$ is the additive identity,
\[
0\cdot a=0.
\]
\hyperref[prf:left-zero-absorption]{\textit{Go to proof.}}
\end{proposition}
\end{propositionbox}

\begin{remark*}[Standard quantified statement]
\[
\operatorname{Distributive}(\cdot,+,A)
\land
\operatorname{Identity}(0,A,+)
\Longrightarrow
\forall a\in A,\ 0\cdot a=0.
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{LeftAbsorbingElement}(0,A,\cdot).
\]
\end{remark*}

\begin{remark*}[Interpretation]
The additive identity becomes a left absorbing element for multiplication.
\end{remark*}

\begin{definitionbox}{Definition (Zero Divisor)}
\begin{definition}[Zero Divisor]
\label{def:zero-divisor}
Let $K$ be an arithmetic system with multiplication and zero. A nonzero
element $a\in K$ is a \emph{zero divisor} if there exists a nonzero element
$b\in K$ such that $ab=0$.
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
a\in K,\ a\ne 0,\ \exists b\in K,\ b\ne 0 \land ab=0.
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{ZeroDivisor}(a,K,\cdot,0).
\]
\end{remark*}

\begin{remark*}[Interpretation]
A zero divisor is a nonzero element that annihilates another nonzero element.
\end{remark*}

\begin{theorembox}{Theorem (No Zero Divisors)}
\begin{theorem}[No Zero Divisors]
\label{thm:no-zero-divisors}
In an arithmetic system with no nonzero zero divisors,
\[
a\cdot b=0 \Longrightarrow a=0 \lor b=0.
\]
\hyperref[prf:no-zero-divisors]{\textit{Go to proof.}}
\end{theorem}
\end{theorembox}

\begin{remark*}[Standard quantified statement]
\[
\operatorname{NoNonzeroZeroDivisors}(K,\cdot,0)
\Longrightarrow
\forall a,b\in K,\ a\cdot b=0 \Rightarrow a=0 \lor b=0.
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{NoZeroDivisors}(K,\cdot,0).
\]
\end{remark*}

\begin{remark*}[Interpretation]
The absence of nonzero zero divisors turns a zero product into a zero factor
alternative.
\end{remark*}
