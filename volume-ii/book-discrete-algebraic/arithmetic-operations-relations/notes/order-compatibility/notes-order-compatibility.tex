\begin{definitionbox}{Definition (Right Monotony)}
\begin{definition}[Right Monotony]
\label{def:right-monotony}
Right monotony means
\[
yRz \Longrightarrow (x\mathbin{O}y)R(x\mathbin{O}z),
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\forall x,y,z\in A,\ yRz \Longrightarrow (x\mathbin{O}y)R(x\mathbin{O}z).
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{RightMonotone}(O,R,A).
\]
\end{remark*}

\begin{remark*}[Interpretation]
Right monotony preserves the relation after a common left input is applied.
\end{remark*}

\begin{definitionbox}{Definition (Positive Cone)}
\begin{definition}[Positive cone]
\label{def:positive-cone}
Let $K$ be an arithmetic system with addition, multiplication, and zero. A
subset $P\subset K$ is called a \emph{positive cone} if it satisfies:
\begin{enumerate}
  \item if $a,b\in P$, then $a+b\in P$;
  \item if $a,b\in P$, then $ab\in P$;
  \item for every $a\in K$, exactly one of the following holds:
  \[
  a\in P,
  \qquad
  a=0,
  \qquad
  -a\in P.
  \]
\end{enumerate}
The strict order associated to $P$ is
\[
a<b \iff b-a\in P.
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
P\subset K,\quad
\forall a,b\in P,\ a+b\in P,\quad
\forall a,b\in P,\ ab\in P,
\]
\[
\forall a\in K,\ \text{exactly one of }a\in P,\ a=0,\ -a\in P\text{ holds}.
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{PositiveCone}(P,K,+,\cdot,0).
\]
\end{remark*}

\begin{remark*}[Interpretation]
A positive cone encodes a strict order through closure under addition and
multiplication and a sign trichotomy.
\end{remark*}
