\begin{definitionbox}{Definition (Reflexive Relation)}
\begin{definition}[Reflexive Relation]
\label{def:reflexive-relation}
\[
\forall x,\ xRx.
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\forall x\in A,\ xRx.
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{Reflexive}(R,A).
\]
\end{remark*}

\begin{remark*}[Interpretation]
Reflexivity says every element is related to itself.
\end{remark*}

\begin{definitionbox}{Definition (Symmetric Relation)}
\begin{definition}[Symmetric Relation]
\label{def:symmetric-relation}
\[
\forall x,y,\ xRy \Longrightarrow yRx.

\end{definitionbox}

\begin{remark*}[Standard quantified statement]

\forall x,y\in A,\ xRy \Longrightarrow yRx.

\operatorname{Symmetric}(R,A).
\]
\end{remark*}

\begin{remark*}[Interpretation]
Symmetry says the relation can be reversed whenever it holds.

\begin{definitionbox}{Definition (Transitive Relation)}
\begin{definition}[Transitive Relation]
\label{def:transitive-relation}
\[
\forall x,y,z,\ (xRy\land yRz)\Longrightarrow xRz.
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\forall x,y,z\in A,\ (xRy\land yRz)\Longrightarrow xRz.

\begin{remark*}[Predicate reading]
\[
\operatorname{Transitive}(R,A).
\]
\end{remark*}

\begin{remark*}[Interpretation]
Transitivity lets two composable relation steps collapse to one step.
\end{remark*}

\begin{definitionbox}{Definition (Trichotomy)}
\begin{definition}[Trichotomy]
\label{def:trichotomy}
The relation $<$ satisfies \emph{trichotomy} if exactly one of
\[
x<y,\qquad x=y,\qquad y<x
\]
holds for every pair $x,y$.
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\forall x,y\in A,\ \text{exactly one of }x<y,\ x=y,\ y<x\text{ holds}.
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{Trichotomous}(<,A).
\]
\end{remark*}

\begin{remark*}[Interpretation]
Trichotomy packages comparability and exclusivity for a strict order.
\end{remark*}
