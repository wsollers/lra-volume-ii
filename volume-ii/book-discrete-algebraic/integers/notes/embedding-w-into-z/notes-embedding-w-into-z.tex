\subsection*{\texorpdfstring{Embedding $\mathbb{W}$ into $\mathbb{Z}$}{Embedding W into Z}}

\begin{definitionbox}{Definition}
\begin{definition}[Embedding of $\mathbb{W}$ into $\mathbb{Z}$]
\label{def:canonical-embedding-w-into-z}
Define
\[
\iota:\mathbb{W}\to\mathbb{Z},
\qquad
\iota(n):=[n,0].
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\iota:\mathbb{W}\to\mathbb{Z},
\qquad
\forall n\in\mathbb{W},\ \iota(n)=[n,0].
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{CanonicalEmbedding}(\iota,\mathbb{W},\mathbb{Z}).
\]
\end{remark*}

\begin{remark*}[Interpretation]
The canonical embedding sends each whole number to its corresponding integer
quotient element.
\end{remark*}

\begin{theorembox}{Theorem}

\label{thm:embedding-w-into-z-injective}
