\begin{remark*}[Exposition]
Tao's construction presents the integers as formal difference expressions.
The placeholder notation \(a {-\!\!-} b\) is intentionally not subtraction:
it records an ordered pair of natural numbers before subtraction has been
constructed. The equality relation identifies two such placeholders exactly
when cross-addition gives the same natural number.
\end{remark*}

\begin{definition}[Tao integer expression]
\label{def:tao-integer-expression}
An integer expression in Tao's construction is a placeholder \(a {-\!\!-} b\)
with \(a,b\in\mathbb{N}\). Two placeholders are equal when
\[
  a {-\!\!-} b = c {-\!\!-} d
  \quad\Longleftrightarrow\quad
  a+d=c+b.
\]
The collection of equivalence classes is denoted \(\mathbb{Z}\).
\end{definition}
\SourceVariantOf{def:integer-prepair}{Tao}{Analysis I, Section 4.1}{source_variant_of}
\SourceVariantOf{def:formal-difference-equivalence}{Tao}{Analysis I, Section 4.1}{source_variant_of}

\begin{remark*}[Standard quantified statement]
\[
  \text{Tao's integer construction is the quotient of }
  \mathbb{N}\times\mathbb{N}
  \text{ by the cross-addition relation.}
\]
\end{remark*}

\begin{remark*}[Interpretation]
The definition replaces the circular phrase \(a-b\) with a formal pair and an
equality criterion that only uses arithmetic already available in
\(\mathbb{N}\). The construction pays the equivalence-relation cost once and
then supports uniform definitions of operations on classes.
\end{remark*}

\begin{remark*}[Notation]
The symbol \(--\) is scaffolding notation. After subtraction is defined on
\(\mathbb{Z}\), the expression \(a {-\!\!-} b\) agrees with the genuine
difference \(a-b\).
\end{remark*}

\begin{remark*}[Exposition]
This presentation contrasts with the signed-natural description of integers.
The signed-natural picture is intuitive, but it forces repeated case splits
for addition, multiplication, and order. The quotient construction centralizes
those checks in the equivalence relation and well-definedness lemmas.
\end{remark*}

\DefinitionalRoot

\begin{definition}[Tao integer operations]
\label{def:tao-integer-operations}
For Tao integer expressions, addition and multiplication are defined by
\[
  (a {-\!\!-} b) + (c {-\!\!-} d) := (a+c) {-\!\!-} (b+d)
\]
and
\[
  (a {-\!\!-} b)(c {-\!\!-} d) := (ac+bd) {-\!\!-} (ad+bc).
\]
\end{definition}

\SourceVariantOf{def:multiplication-on-z-classes}{Tao}{Analysis I, Section 4.1}{source_variant_of}

\begin{remark*}[Standard quantified statement]
\[
  \text{Addition and multiplication on Tao integer representatives are the
  operations forced by formal differences.}
\]
\end{remark*}

\begin{remark*}[Interpretation]
The addition formula adds the positive coordinates and negative coordinates
separately. The multiplication formula expands \((a-b)(c-d)\) and then
collects the positive terms \(ac+bd\) and negative terms \(ad+bc\).
\end{remark*}

\begin{lemma}[Tao 4.1.3 --- Addition and multiplication are well-defined]
\label{lem:int-well-defined}
Let \(a,b,a',b',c,d\in\mathbb{N}\). If
\((a {-\!\!-} b)=(a' {-\!\!-} b')\), then
\begin{align*}
  (a {-\!\!-} b) + (c {-\!\!-} d)
    &= (a' {-\!\!-} b') + (c {-\!\!-} d), \\
  (a {-\!\!-} b)(c {-\!\!-} d)
    &= (a' {-\!\!-} b')(c {-\!\!-} d),
\end{align*}
and similarly when the second representative is replaced by an equal one.
\hyperref[prf:int-well-defined]{Go to proof.}
\end{lemma}

\SourceVariantOf{thm:multiplication-on-z-well-defined}{Tao}{Analysis I, Section 4.1}{reduces_to}

\begin{remark*}[Standard quantified statement]
\[
  \text{Equal representatives give equal sums and products in Tao's
  integer construction.}
\]
\end{remark*}

\begin{remark*}[Interpretation]
The operations must be functions on integer classes, not functions on the
chosen representatives. This lemma supplies the substitution check needed
before the formulas define arithmetic on \(\mathbb{Z}\).
\end{remark*}

\begin{definition}[Tao negation]
\label{def:tao-negation}
For a Tao integer expression, negation is defined by
\[
  -(a {-\!\!-} b) := b {-\!\!-} a.
\]
In particular, if \(n\) is a positive natural number, then
\[
  -n := 0 {-\!\!-} n.
\]
\end{definition}
\SourceVariantOf{def:negation-on-z-classes}{Tao}{Analysis I, Section 4.1}{source_variant_of}

\begin{remark*}[Standard quantified statement]
\[
  \text{Negation swaps the two coordinates of a Tao integer representative.}
\]
\end{remark*}

\begin{remark*}[Interpretation]
Swapping the coordinates changes the formal difference \(a-b\) into the
formal difference \(b-a\). The definition is natural, but like the binary
operations it must be checked against the equality relation.
\end{remark*}

\begin{lemma}[Tao 4.1.5 --- Trichotomy of integers]
\label{lem:int-trichotomy}
Let \(x\in\mathbb{Z}\). Exactly one of the following holds:
\begin{enumerate}
  \item \(x=0\).
  \item \(x=n\) for some positive natural number \(n\).
  \item \(x=-n\) for some positive natural number \(n\).
\end{enumerate}
In the second case \(x\) is positive, and in the third case \(x\) is negative.
\hyperref[prf:int-trichotomy]{Go to proof.}
\end{lemma}
\SourceVariantOf{thm:trichotomy-on-z}{Tao}{Analysis I, Section 4.1}{reduces_to}

\begin{remark*}[Standard quantified statement]
\[
  \text{Every Tao integer is uniquely zero, positive, or negative.}
\]
\end{remark*}

\begin{remark*}[Interpretation]
The equivalence-class construction still recovers the familiar sign
classification. Existence follows by comparing the two natural-number
coordinates of a representative, and uniqueness follows from natural-number
cancellation and trichotomy.
\end{remark*}

\subsection*{Algebraic Laws}

\begin{remark*}[Exposition]
Tao 4.1.6 records the algebraic laws for the constructed integer system:
addition is commutative and associative, \(0\) is an additive identity, every
integer has an additive inverse, multiplication is commutative and associative,
\(1\) is a multiplicative identity, and multiplication distributes over
addition on both sides.
The proposition is the algebraic payoff of the quotient construction. Once
well-definedness is known, the usual laws reduce to the corresponding
arithmetic identities for natural numbers.
\end{remark*}

\begin{definition}[Subtraction]
\label{def:int-subtraction}
For integers \(x,y\in\mathbb{Z}\),
\[
  x-y := x+(-y).
\]
\end{definition}
\SourceVariantOf{def:subtraction-on-z}{Tao}{Analysis I, Section 4.1}{source_variant_of}

\begin{remark*}[Standard quantified statement]
\[
  \text{Subtraction in } \mathbb{Z} \text{ is addition of the additive inverse.}
\]
\end{remark*}

\begin{remark*}[Interpretation]
This definition completes the promise made by the placeholder notation. For
natural numbers \(a,b\), the genuine integer subtraction \(a-b\) agrees with
the class represented by \(a {-\!\!-} b\).
\end{remark*}

\begin{proposition}[Tao 4.1.8 --- No zero divisors]
\label{prop:int-no-zero-div}
Let \(a,b\in\mathbb{Z}\). If \(ab=0\), then \(a=0\) or \(b=0\).
\hyperref[prf:int-no-zero-div]{Go to proof.}
\end{proposition}

  \text{A product of Tao integers is zero only when one factor is zero.}
\]
\end{remark*}

\begin{remark*}[Interpretation]
This is the integral-domain property for the constructed integer system. It
uses trichotomy to reduce the question to positive and negative cases.
\end{remark*}

\begin{corollary}[Tao 4.1.9 --- Cancellation law for \(\mathbb{Z}\)]
\label{cor:int-cancel}
Let \(a,b,c\in\mathbb{Z}\). If \(ac=bc\) and \(c\neq 0\), then \(a=b\).
\hyperref[prf:int-cancel]{Go to proof.}
\end{corollary}
\SourceVariantOf{cor:multiplicative-cancellation-on-z}{Tao}{Analysis I, Section 4.1}{reduces_to}

\begin{remark*}[Standard quantified statement]
\[
  \text{A nonzero integer factor may be cancelled from an equality of products.}
\]
\end{remark*}

\begin{remark*}[Interpretation]
Cancellation follows by rewriting \(ac=bc\) as \((a-b)c=0\) and applying the
absence of zero divisors. It is one of the main algebraic advantages of the
integer construction.
\end{remark*}

\begin{definition}[Tao order on \(\mathbb{Z}\)]
\label{def:tao-order-on-z}
For integers \(n,m\in\mathbb{Z}\),
\[
  n\geq m \quad\Longleftrightarrow\quad n=m+a
  \text{ for some } a\in\mathbb{N},
\]
and
\[
  n>m \quad\Longleftrightarrow\quad n\geq m \text{ and } n\neq m.
\]
\end{definition}
\SourceVariantOf{def:order-on-z}{Tao}{Analysis I, Section 4.1}{source_variant_of}

\begin{remark*}[Standard quantified statement]
\[
  \text{The Tao order on } \mathbb{Z}
  \text{ is defined by translation by natural numbers.}
\]
\end{remark*}

\begin{remark*}[Interpretation]
This order extends the natural-number order through the embedding
\(n\mapsto n {-\!\!-} 0\). It defines comparison by asking whether one
integer is obtained from another by adding a natural number.
\end{remark*}

\subsection*{Order Laws}

\begin{remark*}[Exposition]
Induction does not transfer from \(\mathbb{N}\) to \(\mathbb{Z}\). A property
can hold at \(0\) and be preserved by successor while still failing at negative
integers, because the integer order has no least element.
\end{remark*}

\begin{lemma}[Tao 4.1.11 --- Properties of order on \(\mathbb{Z}\)]
\label{lem:int-order}
Let \(a,b,c\in\mathbb{Z}\). Then:
\begin{enumerate}
  \item \(a>b\) if and only if \(a-b\) is a positive natural number.
  \item If \(a>b\), then \(a+c>b+c\).
  \item If \(a>b\) and \(c\) is a positive integer, then \(ac>bc\).
  \item If \(a>b\), then \(-a<-b\).
  \item If \(a>b\) and \(b>c\), then \(a>c\).
  \item Exactly one of \(a>b\), \(a<b\), and \(a=b\) holds.
\end{enumerate}
\hyperref[prf:int-order]{Go to proof.}
\end{lemma}

\SourceVariantOf{thm:addition-preserves-order-on-z}{Tao}{Analysis I, Section 4.1}{reduces_to}
\SourceVariantOf{thm:positive-multiplication-preserves-order-on-z}{Tao}{Analysis I, Section 4.1}{reduces_to}

\begin{remark*}[Standard quantified statement]
\[
  \text{The Tao order on } \mathbb{Z}
  \text{ has the expected translation, multiplication, reversal,
  transitivity, and trichotomy properties.}
\]
\end{remark*}

\begin{remark*}[Interpretation]
Part (a) converts order into positivity. The remaining properties then follow
by translating order claims into statements about positive natural numbers and
using the algebraic laws already established for \(\mathbb{Z}\).
\end{remark*}
