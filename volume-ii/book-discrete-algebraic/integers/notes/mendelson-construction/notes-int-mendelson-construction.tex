% The Integers --- Mendelson Construction

\begin{remark*}[Exposition]
Mendelson constructs the integers from ordered pairs of positive integers.
The construction has the same quotient shape as Tao's construction, but it
uses equivalence classes \([(n,j)]\) instead of placeholder differences. It
then develops more abstract algebra before returning to the ordered integer
system.
\end{remark*}

\begin{definition}[Mendelson equivalence relation on positive pairs]
\label{def:men-equivalence-relation}
For positive integers \(n,j,k,i\), define
\[
  (n,j)\sim(k,i)
  \quad\Longleftrightarrow\quad
  n+i=k+j.
\]
\end{definition}
\SourceVariantOf{def:integer-prepair}{Mendelson}{Number Systems and the Foundations of Analysis, Chapter 2}{source_variant_of}


\SourceVariantOf{def:formal-difference-equivalence}{Mendelson}{Number Systems and the Foundations of Analysis, Chapter 2}{source_variant_of}

\LeanFormalizes{def:men-equivalence-relation}{lra-lean}{LRA.VolumeII.Integers.Mendelson.Construction.Model}{equivalent}{pending}

\begin{remark*}[Standard quantified statement]
\[
  \text{Mendelson's relation identifies two positive-integer pairs when
  cross-addition agrees.}
\]
\end{remark*}

\begin{remark*}[Interpretation]
The pair \((n,j)\) represents the formal difference \(n-j\). The defining
condition uses only addition of positive integers and therefore avoids
assuming subtraction before the integer system exists.
\end{remark*}

\label{thm:men-equiv}

\begin{definition}[Mendelson integers]
\label{def:men-integers}
\(\mathbb{Z}\) is the set of all equivalence classes of positive-integer
pairs under \(\sim\). The distinguished elements are
\[
  0_{\mathbb{Z}}=[(1,1)],
  \qquad
  1_{\mathbb{Z}}=[(2,1)].
\]
\end{definition}

\begin{definition}[Ring and integral domain]
\label{def:men-ring-integral-domain}
A ring is a set with addition and multiplication such that addition is an
abelian group operation, multiplication is associative, and multiplication
distributes over addition. An integral domain is a commutative ring with unit,
\(0\neq 1\), and no zero divisors.
\end{definition}

\begin{remark*}[Standard quantified statement]
\[
  \text{An integral domain is a commutative ring with identity, nontrivial
  zero and one, and no zero divisors.}
\]
\end{remark*}

\begin{remark*}[Interpretation]
Mendelson inserts abstract algebra here so that properties proved for
\(\mathbb{Z}\) can be reused in later number systems.
\end{remark*}

\begin{definition}[Ordered integral domain]
\label{def:men-ordered-integral-domain}
An ordered integral domain is an integral domain equipped with an order \(<\)
that is trichotomous and transitive, is compatible with addition, and is
compatible with multiplication by positive elements.
\end{definition}

\begin{remark*}[Standard quantified statement]

  compatible with addition and positive multiplication.}
\]
\end{remark*}

\begin{remark*}[Interpretation]
Mendelson proves order facts abstractly and then instantiates them for
\(\mathbb{Z}\). This makes the order theory portable to later ordered
number systems.

\begin{theorem}[Mendelson 4.3 --- Positivity set construction]
\label{thm:men-pos-set}
Let \(R\) be an integral domain with \(0\neq 1\). If
\(\mathcal{P}\subseteq R\) excludes \(0\), trichotomizes each element by
membership in \(\mathcal{P}\), equality to \(0\), or membership of its
negative, and is closed under addition and multiplication, then
\[
  x<y \quad\Longleftrightarrow\quad y-x\in\mathcal{P}
\]
defines an ordered integral domain whose positive set is \(\mathcal{P}\).
\hyperref[prf:men-pos-set]{Go to proof.}
\end{theorem}
\SourceVariantOf{def:positive-integer}{Mendelson}{Number Systems and the Foundations of Analysis, Chapter 2}{reduces_to}
\SourceVariantOf{def:order-on-z}{Mendelson}{Number Systems and the Foundations of Analysis, Chapter 2}{reduces_to}

\begin{remark*}[Standard quantified statement]
\[
  \text{A suitable positive cone on an integral domain determines an ordered
  integral domain.}
\]
\end{remark*}

\begin{remark*}[Interpretation]
The theorem reduces order construction to checking positivity conditions. This
is how Mendelson orders the integers without verifying every order axiom
directly.
\end{remark*}

integers \(q,r\) such that \(\beta=q\alpha+r\) and \(0\leq r<\alpha\).
\hyperref[prf:men-euclidean-division]{Go to proof.}
\end{theorem}

\begin{definition}[Mendelson divisibility]
\label{def:men-divisibility}
For integers \(\alpha,\beta\),
\[
  \alpha\mid\beta
  \quad\Longleftrightarrow\quad
  \text{there exists }\gamma\in\mathbb{Z}\text{ such that }\beta=\alpha\gamma.
\]
\end{definition}
\SourceVariantOf{def:divisibility-in-z}{Mendelson}{Number Systems and the Foundations of Analysis, Chapter 2}{source_variant_of}

\begin{remark*}[Standard quantified statement]
\[
  \text{An integer divides another when the second is an integer multiple of
  the first.}
\]
\end{remark*}

\begin{remark*}[Interpretation]
Divisibility is the basic predicate for the subsequent gcd, prime, and
factorization results.
\end{remark*}

\begin{theorem}[Mendelson 5.2 --- Basic divisibility properties]

\begin{theorem}[Mendelson 5.3 --- GCD and Bezout form]
\label{thm:men-gcd-bezout}
For integers \(\alpha\) and \(\beta\), not both zero, a greatest common divisor
exists. Moreover, the greatest common divisor may be written in Bezout form
\[
  d = x\alpha+y\beta
\]
for some integers \(x,y\).
\hyperref[prf:men-gcd-bezout]{Go to proof.}
\end{theorem}

\begin{theorem}[Mendelson 5.4 --- Relative primeness criterion]
\label{thm:men-relative-primeness}
Two integers are relatively prime if and only if their greatest common divisor
is a unit, equivalently if and only if \(1\) is an integer linear combination
of the two integers up to sign.
\hyperref[prf:men-relative-primeness]{Go to proof.}
\end{theorem}

\begin{remark*}[Standard quantified statement]
\[
  \text{Relative primeness is characterized by a unit gcd and by a Bezout
  linear combination.}
\]
\end{remark*}

\begin{remark*}[Interpretation]
This is the form of relative primeness used in divisibility arguments. It
turns an absence of common divisors into an explicit linear combination.
\end{remark*}

\begin{theorem}[Mendelson 5.6 --- Euclid's lemma]
\label{thm:men-euclid-lemma}
If \(\rho\) is prime and \(\rho\mid\alpha\beta\), then
\[
  \rho\mid\alpha
  \quad\text{or}\quad
  \rho\mid\beta.
\]
\hyperref[prf:men-euclid-lemma]{Go to proof.}
\end{theorem}

\begin{remark*}[Standard quantified statement]
\[
  \text{A prime divisor of a product divides at least one factor.}
\]
\end{remark*}

\begin{remark*}[Interpretation]
Euclid's lemma is the key irreducibility property that makes prime
factorization unique.
\end{remark*}

\begin{theorem}[Mendelson 5.7--5.9 --- Unique prime factorization]
\label{thm:men-unique-prime-factorization}
Every nonzero nonunit integer factors into primes, and this factorization is
unique up to reordering and multiplication of factors by units.
\hyperref[prf:men-unique-prime-factorization]{Go to proof.}
\end{theorem}

\end{remark*}

\begin{remark*}[Interpretation]

Existence follows by descent, while uniqueness is controlled by Euclid's
lemma.

\begin{definition}[Well-ordered integral domain]

\begin{theorem}[Mendelson 9.8 --- Uniqueness of \(\mathbb{Z}\)]
\label{thm:men-well-ordered-integral-domain-uniqueness}
Any two well-ordered integral domains are isomorphic. In particular,
\(\mathbb{Z}\) is unique up to isomorphism as a well-ordered integral domain.
\hyperref[prf:men-well-ordered-integral-domain-uniqueness]{Go to proof.}
\end{theorem}

  to isomorphism.}
\]
\end{remark*}

\begin{remark*}[Interpretation]

only constructed; they are characterized up to the appropriate notion of
structure-preserving equivalence.
\end{remark*}
