\subsection*{\texorpdfstring{Addition on $\mathbb{Z}$-Classes}{Addition on Z-Classes}}

\begin{definitionbox}{Definition}
\begin{definition}[Addition on $\mathbb{Z}$-Classes]
\label{def:addition-on-z-classes}
Define addition on representatives by
\[
[a,b]+[c,d]:=[a+c,b+d].
\]
\end{definition}
\end{definitionbox}

\begin{remark*}[Standard quantified statement]
\[
\forall a,b,c,d\in\mathbb{W},\quad
[a,b]+[c,d]=[a+c,b+d].
\]
\end{remark*}

\begin{remark*}[Predicate reading]
\[
\operatorname{InducedBinaryOperation}(+_{\mathbb{Z}},+_{\mathbb{W}},\mathbb{Z}).
\]
\end{remark*}

\begin{remark*}[Interpretation]
Integer addition is first specified on representatives before passing to the
quotient.
\end{remark*}
